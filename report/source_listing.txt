"""An AVL sequence of arcs, with O(1) links to neighbouring arcs.

Nodes have stable identities: deletion transplants the actual successor node
rather than copying a payload, because queued events refer to arc objects.
"""

from dataclasses import dataclass

from .geometry import breakpoint_x


@dataclass(eq=False, slots=True)
class Arc:
    site: int
    left: "Arc | None" = None
    right: "Arc | None" = None
    parent: "Arc | None" = None
    prev: "Arc | None" = None
    next: "Arc | None" = None
    height: int = 1
    generation: int = 0
    alive: bool = True


def height(node):
    return node.height if node else 0


def update(node):
    node.height = 1 + max(height(node.left), height(node.right))


class BeachLine:
    def __init__(self):
        self.root = None
        self.first = None
        self.count = 0
        self.max_height = 0

    def _replace(self, old, new):
        parent = old.parent
        if parent is None:
            self.root = new
        elif parent.left is old:
            parent.left = new
        else:
            parent.right = new
        if new:
            new.parent = parent

    def _left(self, node):
        pivot = node.right
        self._replace(node, pivot)
        node.right = pivot.left
        if node.right:
            node.right.parent = node
        pivot.left, node.parent = node, pivot
        update(node)
        update(pivot)
        return pivot

    def _right(self, node):
        pivot = node.left
        self._replace(node, pivot)
        node.left = pivot.right
        if node.left:
            node.left.parent = node
        pivot.right, node.parent = node, pivot
        update(node)
        update(pivot)
        return pivot

    def _rebalance(self, node):
        while node:
            update(node)
            balance = height(node.left) - height(node.right)
            if balance > 1:
                if height(node.left.left) < height(node.left.right):
                    self._left(node.left)
                node = self._right(node)
            elif balance < -1:
                if height(node.right.right) < height(node.right.left):
                    self._right(node.right)
                node = self._left(node)
            node = node.parent
        self.max_height = max(self.max_height, height(self.root))

    def insert_after(self, before, new):
        if not new.alive or new.parent or new.left or new.right:
            raise ValueError("Insert a fresh unattached arc")
        if before is None:
            if self.root:
                anchor = self.first
                anchor.left = new
                new.parent = anchor
                new.next, anchor.prev, self.first = anchor, new, new
            else:
                self.root = self.first = new
        else:
            if not before.alive:
                raise ValueError("Insertion anchor is not live")
            anchor = before
            if before.right:
                anchor = before.right
                while anchor.left:
                    anchor = anchor.left
                anchor.left = new
            else:
                anchor.right = new
            new.parent = anchor
            new.prev, new.next = before, before.next
            if before.next:
                before.next.prev = new
            before.next = new
        self.count += 1
        self._rebalance(new)

    def remove(self, node):
        if not node.alive:
            raise ValueError("Arc has already been removed")
        if node.prev:
            node.prev.next = node.next
        else:
            self.first = node.next
        if node.next:
            node.next.prev = node.prev
        if node.left and node.right:
            successor = node.right
            while successor.left:
                successor = successor.left
            if successor.parent is node:
                start = successor
            else:
                start = successor.parent
                self._replace(successor, successor.right)
                successor.right = node.right
                successor.right.parent = successor
            self._replace(node, successor)
            successor.left = node.left
            successor.left.parent = successor
            update(successor)
        else:
            start = node.parent
            self._replace(node, node.left or node.right)
        node.alive = False
        node.generation += 1
        node.left = node.right = node.parent = None
        self.count -= 1
        self._rebalance(start)

    def locate(self, x, directrix, points):
        node = self.root
        while node:
            if node.prev and x < breakpoint_x(
                points[node.prev.site], points[node.site], directrix
            ):
                node = node.left
            elif node.next and x > breakpoint_x(
                points[node.site], points[node.next.site], directrix
            ):
                node = node.right
            else:
                return node
        raise ArithmeticError("Beach-line search escaped its ordered arcs")

    def __iter__(self):
        node = self.first
        while node:
            yield node
            node = node.next

    def validate(self):
        """Expensive debug invariant check; deliberately not in production loops."""
        order = []

        def walk(node, parent):
            if node is None:
                return 0
            assert node.alive and node.parent is parent
            lh = walk(node.left, node)
            order.append(node)
            rh = walk(node.right, node)
            assert abs(lh - rh) <= 1
            assert node.height == 1 + max(lh, rh)
            return node.height

        walk(self.root, None)
        assert len(order) == self.count
        assert list(self) == order
        for i, arc in enumerate(order):
            assert arc.prev is (order[i - 1] if i else None)
            assert arc.next is (order[i + 1] if i + 1 < len(order) else None)
        return True

"""Circle-event eligibility for three consecutive arcs in a downward sweep."""

import math

from .geometry import circle_bottom, circumcircle, orient


def candidate_circle(a, b, c, sweep_y):
    if a == c or a == b or b == c or orient(a, b, c) >= 0:
        return None
    circle = circumcircle(a, b, c)
    if circle is None:
        return None
    centre, _ = circle
    bottom = circle_bottom(centre, b)
    # Permit same-level events (essential for cocircular sites), but never a
    # materially earlier event. The allowance is tied to arithmetic precision.
    allowance = 64 * math.ulp(max(1.0, abs(sweep_y)))
    if bottom > sweep_y + allowance:
        return None
    return centre, min(bottom, sweep_y)

"""Documented batch and interactive entry points."""

import argparse
import json
import sys

from .diagram import construct
from .geometry import Box
from .io import random_sites, read_sites, write_csv, write_json


def main(argv=None):
    argv = sys.argv[1:] if argv is None else argv
    parser = argparse.ArgumentParser(
        description="Fortune sweep-line Voronoi diagrams, from scratch."
    )
    group = parser.add_mutually_exclusive_group()
    group.add_argument("--input", help="UTF-8 coordinate file: x y or x,y per line")
    group.add_argument(
        "--random", type=int, metavar="N", help="N uniform sites in the unit square"
    )
    parser.add_argument("--seed", type=int, default=2026)
    parser.add_argument(
        "--bbox", type=float, nargs=4, metavar=("XMIN", "YMIN", "XMAX", "YMAX")
    )
    parser.add_argument(
        "--output", help="Batch PNG/SVG path, or standalone HTML with Canvas --animate"
    )
    parser.add_argument(
        "--csv", help="Ordered cell boundaries and explicit clipped-edge records"
    )
    parser.add_argument(
        "--json",
        dest="json_path",
        help="Complete diagram including rays and defining triples",
    )
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument(
        "--interactive", action="store_true", help="Click-to-add interface"
    )
    mode.add_argument(
        "--animate",
        action="store_true",
        help="Replay the sweep with Play/Pause and Next event (up to 200 sites)",
    )
    parser.add_argument(
        "--animation-backend",
        choices=("canvas", "matplotlib"),
        default="canvas",
        help="Canvas pointer replay (default), or the Matplotlib event inspector",
    )
    parser.add_argument(
        "--show", action="store_true", help="Display batch output on screen"
    )
    parser.add_argument(
        "--no-labels", action="store_true", help="Hide indices for dense previews"
    )
    args = parser.parse_args(argv)
    try:
        sites = (
            read_sites(args.input)
            if args.input
            else random_sites(args.random, args.seed)
            if args.random is not None
            else []
        )
        box = (
            Box(*args.bbox)
            if args.bbox
            else Box(0, 0, 1, 1)
            if args.random is not None
            else None
        )
        if args.animate:
            if args.csv or args.json_path:
                parser.error(
                    "Use batch mode for CSV/JSON; Canvas replay --output saves HTML."
                )
            if args.animation_backend == "canvas":
                from .replay_canvas import launch
            else:
                from .replay_view import launch

            if not sites and args.input is None and args.random is None:
                sites = random_sites(8, args.seed)
                box = box or Box(0, 0, 1, 1)
            if not sites:
                raise ValueError("At least one finite site is required")
            replay = launch(
                sites,
                box,
                args.output
                or (
                    "replay.html"
                    if args.animation_backend == "canvas"
                    else "replay.png"
                ),
                labels=not args.no_labels,
            )
            if args.animation_backend == "canvas" and hasattr(replay, "path"):
                print(f"Opened pointer replay: {replay.path}")
            return 0
        if args.interactive or not argv:
            from .interactive import launch

            launch(sites, box, args.output or "interactive.png")
            return 0
        if not sites:
            parser.error("Specify --input FILE, --random N, or --interactive")
        diagram = construct(sites, box)
        if args.csv:
            write_csv(diagram, args.csv)
        if args.json_path:
            write_json(diagram, args.json_path)
        if args.output or args.show:
            from .visualize import render

            render(diagram, args.output, show=args.show, labels=not args.no_labels)
        print(json.dumps(diagram.statistics, indent=2))
    except (ValueError, ArithmeticError, OSError) as exc:
        print(f"Error: {exc}", file=sys.stderr)
        return 2
    return 0

"""Bounded half-plane intersection over sweep-generated neighbours only.

Angle sorting plus a deque costs O(k log k) for a cell of degree k. This avoids
the quadratic cost of repeatedly clipping a high-degree polygon. The four box
constraints make every nonempty result bounded.
"""

import math
from collections import deque
from dataclasses import dataclass
from decimal import Decimal, localcontext
from fractions import Fraction
from functools import cmp_to_key

from .geometry import Point


@dataclass(slots=True)
class HalfPlane:
    a: float
    b: float
    c: float  # unit-normal inequality a*x + b*y <= c
    neighbour: int | None = None
    focuses: tuple[Point, Point] | None = None

    def exact(self):
        if self.focuses is None:
            return Decimal(self.a), Decimal(self.b), Decimal(self.c)
        p, q = self.focuses
        px, py, qx, qy = map(Decimal, (p.x, p.y, q.x, q.y))
        a, b = qx - px, qy - py
        return a, b, (a * (px + qx) + b * (py + qy)) / 2

    def rational(self):
        """Exact dyadic coefficients, used only for uncertain topology decisions."""
        if self.focuses is None:
            return Fraction(self.a), Fraction(self.b), Fraction(self.c)
        p, q = self.focuses
        px, py, qx, qy = map(Fraction, (p.x, p.y, q.x, q.y))
        a, b = qx - px, qy - py
        return a, b, (a * (px + qx) + b * (py + qy)) / 2


def parallel(p, q):
    if abs(p.a * q.b - p.b * q.a) > 1e-12:
        return False
    a, b, _ = p.rational()
    d, e, _ = q.rational()
    return a * e - b * d == 0


def angle_compare(p, q):
    hp = 0 if p.b > 0 or (p.b == 0 and p.a >= 0) else 1
    hq = 0 if q.b > 0 or (q.b == 0 and q.a >= 0) else 1
    if hp != hq:
        return hp - hq
    cross = p.a * q.b - p.b * q.a
    if abs(cross) <= 1e-12:
        a, b, _ = p.rational()
        d, e, _ = q.rational()
        cross = a * e - b * d
    return -1 if cross > 0 else 1 if cross < 0 else 0


def tighter(p, q):
    """Whether q is stricter than a parallel, same-direction plane p."""
    a, b, c = p.rational()
    d, e, f = q.rational()
    return c * max(abs(d), abs(e)) > f * max(abs(a), abs(b))


def corner_inside(test, p, q):
    """Test a corner using exact rational signs when the float result is close.

    Comparing rounded corner coordinates with an epsilon can retain an invalid
    support. Instead evaluate the intersection in homogeneous coordinates, so
    the uncertain sign requires no inexact division.
    """
    point = intersection(p, q)
    if point is None:
        return False
    value = test.a * point.x + test.b * point.y - test.c
    if abs(value) > 1e-10 * (1 + abs(test.c) + abs(point.x) + abs(point.y)):
        return value < 0
    a, b, c = p.rational()
    d, e, f = q.rational()
    g, h, k = test.rational()
    determinant = a * e - b * d
    sign = g * (c * e - b * f) + h * (a * f - c * d) - k * determinant
    return sign <= 0 if determinant > 0 else sign >= 0


def concurrent(p, q, r):
    a, b, c = p.rational()
    d, e, f = q.rational()
    g, h, k = r.rational()
    return a * (e * k - f * h) - b * (d * k - f * g) + c * (d * h - e * g) == 0


def intersection(p, q):
    det = p.a * q.b - p.b * q.a
    if abs(det) < 1e-7:
        # Recompute from exact binary focus coordinates, not from rounded
        # normalized plane coefficients: close-site bisectors amplify that error.
        if parallel(p, q):
            return None
        with localcontext() as ctx:
            ctx.prec = 80
            a, b, c = p.exact()
            d, e, f = q.exact()
            denominator = a * e - b * d
            if not denominator:
                return None
            return Point(
                float((c * e - b * f) / denominator),
                float((a * f - c * d) / denominator),
            )
    return Point((p.c * q.b - p.b * q.c) / det, (p.a * q.c - p.c * q.a) / det)


def halfplane_polygon(planes):
    ordered = sorted(planes, key=cmp_to_key(angle_compare))
    unique = []
    for h in ordered:
        if unique:
            p = unique[-1]
            if parallel(p, h) and p.a * h.a + p.b * h.b > 0:
                if tighter(p, h):
                    unique[-1] = h
                continue
        unique.append(h)
    active = deque()
    for h in unique:
        while len(active) > 1:
            p = intersection(active[-2], active[-1])
            if p is None:
                return [], []
            if corner_inside(h, active[-2], active[-1]):
                break
            active.pop()
        while len(active) > 1:
            p = intersection(active[0], active[1])
            if p is None:
                return [], []
            if corner_inside(h, active[0], active[1]):
                break
            active.popleft()
        if active and intersection(active[-1], h) is None:
            return [], []
        active.append(h)
    while len(active) > 2 and not corner_inside(active[0], active[-2], active[-1]):
        active.pop()
    while len(active) > 2 and not corner_inside(active[-1], active[0], active[1]):
        active.popleft()
    if len(active) < 3:
        return [], []
    active = list(active)
    vertices = [intersection(active[i - 1], active[i]) for i in range(len(active))]
    if any(p is None for p in vertices):
        return [], []
    # A short edge can be real: tiny cells must not be removed by a distance
    # cutoff. Discard a support only if three defining lines are truly concurrent.
    active = [
        h
        for i, h in enumerate(active)
        if not (
            vertices[i].distance(vertices[(i + 1) % len(vertices)]) < 1e-12
            and concurrent(active[i - 1], h, active[(i + 1) % len(active)])
        )
    ]
    if len(active) < 3:
        return [], []
    vertices = [intersection(active[i - 1], active[i]) for i in range(len(active))]
    return vertices, active


def clipped_cell(site, neighbours, points, box):
    p = points[site]
    planes = [
        HalfPlane(-1, 0, -box.xmin),
        HalfPlane(1, 0, box.xmax),
        HalfPlane(0, -1, -box.ymin),
        HalfPlane(0, 1, box.ymax),
    ]
    for j in neighbours:
        q = points[j]
        dx, dy = q.x - p.x, q.y - p.y
        d = math.hypot(dx, dy)
        if not d:
            raise ArithmeticError("Duplicate site survived deduplication")
        a, b = dx / d, dy / d
        planes.append(
            HalfPlane(a, b, a * (p.x + q.x) / 2 + b * (p.y + q.y) / 2, j, (p, q))
        )
    return halfplane_polygon(planes)

"""Public construction API and ordered clipped-cell output model."""

import math
from dataclasses import dataclass, field

from .clipping import clipped_cell

@dataclass(slots=True)
class Cell:

    input_indices: list[int]
    vertices: list[Point]
    unbounded: bool
    boundary_neighbours: list[int | None] = field(default_factory=list)


@dataclass(slots=True)
class Edge:

    start: Point
    end: Point
    unbounded: bool
    kind: str
    origin: Point | None
    direction: Point | None


@dataclass(slots=True)
class Diagram:
    sites: list[Point]
    cells: list[Cell]
    edges: list[Edge]
    vertices: list[Vertex]
    box: Box
    statistics: dict = field(default_factory=dict)


def hull_order(points):
    """Convex-hull boundary, retaining collinear sites along a hull edge."""
    ids = sorted(range(len(points)), key=lambda i: (points[i].x, points[i].y))
    lower, upper = [], []
    for i in ids:
        while (
            len(lower) > 1
            and orient(points[lower[-2]], points[lower[-1]], points[i]) < 0
        ):
            lower.pop()
        lower.append(i)
    for i in reversed(ids):
        while (
            len(upper) > 1
            and orient(points[upper[-2]], points[upper[-1]], points[i]) < 0
        ):
            upper.pop()
        upper.append(i)
    return lower[:-1] + upper[:-1]


def construct(coordinates, box=None, *, debug=False):
    """Build a Voronoi diagram; exact duplicates share one cell and index list.

    Geometric calculations use a translated, uniformly scaled frame. Distinct
    near-coincident points are not deliberately merged. Extremely disparate
    coordinates that collapse in normalization are rejected explicitly.
    """
    started = perf_counter()
    sites, groups, index = [], [], {}
    input_count = 0
    for i, value in enumerate(coordinates):
        p = (
            value
            if isinstance(value, Point)
            else Point(float(value[0]), float(value[1]))
        )
        if not math.isfinite(p.x) or not math.isfinite(p.y):
            raise ValueError(f"Site {i} has a non-finite coordinate")
        key = (p.x, p.y)
        if key not in index:
            index[key] = len(sites)
            sites.append(p)
            groups.append([])
        groups[index[key]].append(i)
        input_count += 1
    if not sites:
        raise ValueError("At least one finite site is required")
    xmin, xmax = min(p.x for p in sites), max(p.x for p in sites)
    ymin, ymax = min(p.y for p in sites), max(p.y for p in sites)
    scale = max(xmax - xmin, ymax - ymin)
    if not scale:
        scale = max(1.0, abs(xmin) * 1e-6, abs(ymin) * 1e-6)
    if not math.isfinite(scale):
        raise ValueError("Coordinate extent exceeds supported floating-point range")
    ox, oy = xmin + (xmax - xmin) / 2, ymin + (ymax - ymin) / 2

    def normalize(p):
        return Point((p.x - ox) / scale, (p.y - oy) / scale)

    def world(p):
        return Point(ox + scale * p.x, oy + scale * p.y)

    if box is None:
        pad = scale * 0.1
        box = Box(xmin - pad, ymin - pad, xmax + pad, ymax + pad)
    elif not isinstance(box, Box):
        box = Box(*box)
    nbox = Box(
        (box.xmin - ox) / scale,
        (box.ymin - oy) / scale,
        (box.xmax - ox) / scale,
        (box.ymax - oy) / scale,
    )
    normalized = [normalize(p) for p in sites]
    if len(set(normalized)) != len(sites):
        raise ValueError("Distinct sites collapse at this floating-point scale")
    collinear = all_collinear(sites)
    sweep_start = perf_counter()
    sweep = (
        collinear_neighbours(normalized)
        if collinear
        else FortuneSweep(normalized, debug=debug, original_points=sites).run()
    )
    sweep_seconds = perf_counter() - sweep_start
    hull = [] if collinear else hull_order(sites)
    boundary = set(range(len(sites))) if collinear else set(hull)
    hull_edges = {tuple(sorted((a, b))) for a, b in zip(hull, hull[1:] + hull[:1])}
    # A single-ended trace is a ray only on a hull adjacency. Otherwise it is
    # a zero-length diagonal of a cocircular triangulation, not a real boundary.
    zero_edges = {
        key
        for key, edge in sweep.edges.items()
        if len(edge.vertices) == 1 and key not in hull_edges and not collinear
    }
    vertices = [Vertex(world(v.point), set(v.defining)) for v in sweep.vertices]
    for identity, i in sweep._vertex_index.items():
        vertices[i].point = Point(float(identity[0]), float(identity[1]))
    cells, edges = [], []
    for i in range(len(sites)):
        neighbours = [
            j
            for j in sorted(sweep.neighbours[i])
            if tuple(sorted((i, j))) not in zero_edges
        ]
        polygon, planes = clipped_cell(i, neighbours, normalized, nbox)
        output_polygon = [world(p) for p in polygon]
        p = sites[i]
        if (
            box.xmin <= p.x <= box.xmax
            and box.ymin <= p.y <= box.ymax
            and (len(output_polygon) < 3 or polygon_area(output_polygon) <= 0)
        ):
            raise ArithmeticError(
                f"Site {i}: output precision cannot retain a positive cell"
            )
        cells.append(
            Cell(
                i,
                groups[i],
                output_polygon,
                i in boundary,
                [h.neighbour for h in planes],
            )
        )
        # Plane k supports the edge from vertex k to vertex k+1 (cyclic).
        for k, h in enumerate(planes):
            j = h.neighbour
            if j is None or i >= j:
                continue
            p, q = polygon[k], polygon[(k + 1) % len(polygon)]
            if p == q:
                continue  # zero-length diagonal at a cocircular vertex
            trace = sweep.edges[(i, j)]
            ends = trace.vertices
            kind = "segment" if len(ends) >= 2 else ("ray" if ends else "line")
            origin = vertices[ends[0]].point if ends else None
            direction = None
            if kind == "ray":
                centre = sweep.vertices[ends[0]].point
                endpoint = p if centre.distance(p) > centre.distance(q) else q
                length = centre.distance(endpoint)
                if length:
                    direction = Point(
                        (endpoint.x - centre.x) / length,
                        (endpoint.y - centre.y) / length,
                    )
            elif kind == "line":
                length = p.distance(q)
                direction = Point((q.x - p.x) / length, (q.y - p.y) / length)
            edges.append(
                Edge(
                    (i, j),
                    world(p),
                    world(q),
                    kind != "segment",
                    kind,
                    origin,
                    direction,
                )
            )
    statistics = dict(sweep.statistics)
    statistics.update(
        input_sites=input_count,
        unique_sites=len(sites),
        duplicates=input_count - len(sites),
        cells=len(cells),
        vertices=len(vertices),
        edges=len(edges),
        sweep_seconds=sweep_seconds,
        construction_seconds=perf_counter() - started,
    )
    return Diagram(sites, cells, edges, vertices, box, statistics)

"""Deterministic priority queue: descending y, sites before circles, then x."""

import heapq
import itertools
from dataclasses import dataclass

from .beachline import Arc
from .geometry import Point


@dataclass(slots=True)
class Event:
    y: float
    x: float
    kind: int  # 0 = site, 1 = circle
    site: int | None = None
    arc: Arc | None = None
    generation: int = 0
    centre: Point | None = None
    defining: tuple[int, int, int] | None = None


class EventQueue:
    def __init__(self):
        self._heap = []
        self._serial = itertools.count()
        self.peak_size = 0
        self.pushed = 0

    def push(self, event):
        heapq.heappush(
            self._heap, (-event.y, event.kind, event.x, next(self._serial), event)
        )
        self.pushed += 1
        self.peak_size = max(self.peak_size, len(self._heap))

    def pop(self):
        return heapq.heappop(self._heap)[-1]

    def peek(self):
        """Inspect the next event without changing its heap priority."""
        return self._heap[0][-1] if self._heap else None

    def preview(self, limit=8):
        """Read a bounded, priority-ordered view, including stale entries."""
        if limit < 0:
            raise ValueError("Preview limit must be nonnegative")
        return [item[-1] for item in heapq.nsmallest(limit, self._heap)]

    def __len__(self):
        return len(self._heap)

    def __bool__(self):
        return bool(self._heap)

"""Small geometric primitives, with explicit singular cases and filtered signs."""

import math
from dataclasses import dataclass
from decimal import Decimal, localcontext
from fractions import Fraction


@dataclass(frozen=True, slots=True)
class Point:
    x: float
    y: float

    def distance(self, other: "Point") -> float:
        return math.hypot(self.x - other.x, self.y - other.y)


@dataclass(frozen=True, slots=True)
class Box:
    xmin: float
    ymin: float
    xmax: float
    ymax: float

    def __post_init__(self):
        if not all(math.isfinite(v) for v in self.as_tuple()):
            raise ValueError("Bounding-box coordinates must be finite")
        if self.xmin >= self.xmax or self.ymin >= self.ymax:
            raise ValueError("Bounding box must have positive width and height")

    def as_tuple(self):
        return self.xmin, self.ymin, self.xmax, self.ymax


def orient(a: Point, b: Point, c: Point) -> float:
    """Positive for CCW. Exact dyadic fallback resolves uncertain float products.

    This is a filtered predicate, not arbitrary-precision construction: input
    coordinates and returned circumcentres are still IEEE double precision.
    """
    ux, uy = b.x - a.x, b.y - a.y
    vx, vy = c.x - a.x, c.y - a.y
    left, right = ux * vy, uy * vx
    det = left - right
    if abs(det) > 8e-16 * (abs(left) + abs(right)):
        return det
    ax, ay, bx, by, cx, cy = map(Fraction, (a.x, a.y, b.x, b.y, c.x, c.y))
    exact = (bx - ax) * (cy - ay) - (by - ay) * (cx - ax)
    try:
        value = float(exact)
    except OverflowError:
        value = math.inf if exact > 0 else -math.inf
    if value == 0.0 and exact:
        return math.copysign(math.ulp(0.0), 1 if exact > 0 else -1)
    return value


def circumcircle(a: Point, b: Point, c: Point):
    """Return (centre, radius), or None for exactly collinear input.

    Translation to a local origin avoids subtracting large squared coordinates.
    For cancellation-prone determinants, recompute the construction in Decimal.
    """
    ux, uy = b.x - a.x, b.y - a.y
    vx, vy = c.x - a.x, c.y - a.y
    cross = ux * vy - uy * vx
    # A reliable determinant sign alone does not imply an accurate centre.
    # Ill-conditioned (almost collinear) triples need high-precision coordinates.
    if abs(cross) <= 1e-8 * (abs(ux * vy) + abs(uy * vx)):
        with localcontext() as ctx:
            ctx.prec = 80
            ax, ay = Decimal(a.x), Decimal(a.y)
            u, v = Decimal(b.x) - ax, Decimal(b.y) - ay
            s, t = Decimal(c.x) - ax, Decimal(c.y) - ay
            d = 2 * (u * t - v * s)
            if not d:
                return None
            u2, s2 = u * u + v * v, s * s + t * t
            centre = Point(
                float(ax + (u2 * t - s2 * v) / d), float(ay + (u * s2 - s * u2) / d)
            )
    else:
        u2, v2 = ux * ux + uy * uy, vx * vx + vy * vy
        centre = Point(
            a.x + (u2 * vy - v2 * uy) / (2 * cross),
            a.y + (ux * v2 - vx * u2) / (2 * cross),
        )
    radius = centre.distance(a)
    if not math.isfinite(radius):
        raise ArithmeticError("Circumcircle is outside the supported float range")
    return centre, radius


def parabola_y(focus: Point, x: float, directrix: float) -> float:
    gap = focus.y - directrix
    if gap <= 0:
        raise ValueError("A finite beach parabola needs its focus above the directrix")
    return (x - focus.x) ** 2 / (2 * gap) + (focus.y + directrix) / 2


def breakpoint_x(left: Point, right: Point, directrix: float) -> float:
    """Breakpoint between consecutive arcs in a downward sweep.

    Equal-height sites give a linear equation. A focus on the directrix gives
    a vertical limiting arc. The remaining quadratic is solved with a stable
    two-root formula, in coordinates translated to the left focus.
    """
    if left.y == right.y:
        return left.x + (right.x - left.x) / 2
    dp, dq = left.y - directrix, right.y - directrix
    if dp <= 0:
        return left.x
    if dq <= 0:
        return right.x
    dx = right.x - left.x
    aa = right.y - left.y
    bb = 2 * dp * dx
    cc = -dp * dx * dx - aa * dp * dq
    discr = bb * bb - 4 * aa * cc
    # Roundoff may make a mathematically nonnegative discriminant slightly negative.
    if discr < 0:
        if discr < -2e-14 * (bb * bb + abs(4 * aa * cc)):
            raise ArithmeticError("Negative breakpoint discriminant")
        discr = 0.0
    q = -0.5 * (bb + math.copysign(math.sqrt(discr), bb))
    roots = (q / aa, cc / q) if q else (-bb / (2 * aa),) * 2
    root = min(roots) if left.y > right.y else max(roots)
    return left.x + root


def circle_bottom(centre: Point, through: Point) -> float:
    """Avoid cancellation in centre.y - radius when the centre is high above."""
    dx, dy = centre.x - through.x, centre.y - through.y
    radius = math.hypot(dx, dy)
    if dy >= 0 and radius + dy:
        return through.y - dx * dx / (radius + dy)
    return centre.y - radius


def circle_identity(a, b, c):
    """Exact circle identity for grouping events, not a rounded-centre hash.

    Coordinates are finite input doubles, hence exact dyadic fractions. Keeping
    distinct circles distinct also preserves real tiny edges between vertices.
    """
    ax, ay, bx, by, cx, cy = map(Fraction, (a.x, a.y, b.x, b.y, c.x, c.y))
    ux, uy, vx, vy = bx - ax, by - ay, cx - ax, cy - ay
    denominator = 2 * (ux * vy - uy * vx)
    if not denominator:
        return None
    u2, v2 = ux * ux + uy * uy, vx * vx + vy * vy
    x = ax + (u2 * vy - v2 * uy) / denominator
    y = ay + (ux * v2 - vx * u2) / denominator
    return x, y, (x - ax) ** 2 + (y - ay) ** 2


def polygon_area(points) -> float:
    if len(points) < 3:
        return 0.0
    origin = points[0]
    return (
        abs(
            math.fsum(
                orient(origin, points[i], points[i + 1])
                for i in range(1, len(points) - 1)
            )
        )
        / 2
    )

"""Click-to-add interface with explicit computation, clear and export buttons."""

from .diagram import construct
from .geometry import Box, Point
from .io import write_csv, write_json
from .visualize import draw


class InteractiveController:
    def __init__(self, figure, axes, sites=(), box=None, output="interactive.png"):
        self.figure, self.axes = figure, axes
        self.sites = list(sites)
        self.box = box or Box(0, 0, 1, 1)
        self.output = output
        self.diagram = None
        self.replays = []
        self.status = figure.text(
            0.13,
            0.02,
            "Click to add sites; Compute or double-click to build.",
            fontsize=9,
        )
        self.figure.canvas.mpl_connect("button_press_event", self.click)
        self.clear_axes()

    def clear_axes(self):
        self.axes.clear()
        b = self.box
        self.axes.set(
            xlim=(b.xmin, b.xmax),
            ylim=(b.ymin, b.ymax),
            xlabel="x",
            ylabel="y",
            title="Fortune sweep | click to add points",
        )
        self.axes.set_aspect("equal")
        if self.sites:
            self.axes.scatter(
                [p.x for p in self.sites], [p.y for p in self.sites], c="black"
            )

    def click(self, event):
        if event.inaxes is not self.axes or event.button != 1 or event.xdata is None:
            return
        if event.dblclick:
            self.compute()
            return
        self.sites.append(Point(event.xdata, event.ydata))
        self.diagram = None
        self.clear_axes()
        self.status.set_text(
            f"{len(self.sites)} input sites; compute to update the cells."
        )
        self.figure.canvas.draw_idle()

    def compute(self, event=None):
        if not self.sites:
            self.status.set_text("Add at least one site first.")
            return
        try:
            self.diagram = construct(self.sites, self.box)
            draw(self.diagram, self.axes)
            stats = self.diagram.statistics
            self.status.set_text(
                f"{stats['unique_sites']} cells; {stats['construction_seconds']:.4f} s. Export saves image, CSV and JSON."
            )
        except (ValueError, ArithmeticError) as exc:
            self.status.set_text(str(exc))
        self.figure.canvas.draw_idle()

    def clear(self, event=None):
        self.sites.clear()
        self.diagram = None
        self.clear_axes()
        self.status.set_text("Cleared. Click to add new sites.")
        self.figure.canvas.draw_idle()

    def animate(self, event=None):
        from pathlib import Path

        from .replay_canvas import launch

        if not self.sites:
            self.status.set_text("Add at least one site first.")
        else:
            try:
                path = Path(self.output)
                replay = launch(
                    self.sites,
                    self.box,
                    path.with_name(path.stem + "-replay.html"),
                    block=False,
                )
                self.replays.append(replay)
                self.status.set_text(
                    "Pointer replay opened in the browser. Move over the diagram."
                )
            except (ValueError, ArithmeticError) as exc:
                self.status.set_text(str(exc))
        self.figure.canvas.draw_idle()

    def export(self, event=None):
        from pathlib import Path

        if self.diagram is None:
            self.compute()
        if self.diagram is None:
            return
        path = Path(self.output)
        path.parent.mkdir(parents=True, exist_ok=True)
        self.figure.savefig(path, dpi=160)
        write_csv(self.diagram, path.with_suffix(".csv"))
        write_json(self.diagram, path.with_suffix(".json"))
        self.status.set_text(f"Saved {path.name}, CSV and JSON.")
        self.figure.canvas.draw_idle()


def launch(sites=(), box=None, output="interactive.png"):
    import matplotlib.pyplot as plt
    from matplotlib.widgets import Button

    fig, axes = plt.subplots(figsize=(9, 8))
    fig.subplots_adjust(bottom=0.17)
    controller = InteractiveController(fig, axes, sites, box, output)
    controller.buttons = []
    for x, label, callback in (
        (0.12, "Compute", controller.compute),
        (0.32, "Animate", controller.animate),
        (0.52, "Clear", controller.clear),
        (0.72, "Export", controller.export),
    ):
        button = Button(fig.add_axes([x, 0.075, 0.15, 0.05]), label)
        button.on_clicked(callback)
        controller.buttons.append(button)
    plt.show()
    return controller

"""Coordinate parsing and explicit ordered-cell, clipped-edge exports."""

import csv
import json
import random
from pathlib import Path

from .geometry import Point


def read_sites(path):
    sites = []
    for number, line in enumerate(
        Path(path).read_text(encoding="utf-8-sig").splitlines(), 1
    ):
        line = line.split("#", 1)[0].strip()
        if not line:
            continue
        fields = line.replace(",", " ").split()
        if len(fields) != 2:
            raise ValueError(f"{path}, line {number}: expected exactly x y or x,y")
        try:
            sites.append(Point(float(fields[0]), float(fields[1])))
        except ValueError as exc:
            raise ValueError(f"{path}, line {number}: invalid number") from exc
    return sites


def random_sites(count, seed):
    if count < 1:
        raise ValueError("Random site count must be positive")
    rng = random.Random(seed)
    return [Point(rng.random(), rng.random()) for _ in range(count)]


def ensure_parent(path):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    return path


def write_csv(diagram, path):
    edge_by_pair = {e.sites: e for e in diagram.edges}
    fields = [
        "cell",
        "input_indices",
        "site_x",
        "site_y",
        "unbounded_cell",
        "vertex_order",
        "x",
        "y",
        "next_x",
        "next_y",
        "neighbour",
        "edge_kind",
        "ray_origin_x",
        "ray_origin_y",
        "direction_x",
        "direction_y",
    ]
    with ensure_parent(path).open("w", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields)
        writer.writeheader()
        for c in diagram.cells:
            site = diagram.sites[c.site]
            base = dict(
                cell=c.site,
                input_indices=";".join(map(str, c.input_indices)),
                site_x=site.x,
                site_y=site.y,
                unbounded_cell=int(c.unbounded),
            )
            if not c.vertices:
                writer.writerow(base | {"edge_kind": "empty_clipped_cell"})
            for k, p in enumerate(c.vertices):
                q = c.vertices[(k + 1) % len(c.vertices)]
                j = c.boundary_neighbours[k]
                edge = (
                    edge_by_pair.get((min(c.site, j), max(c.site, j)))
                    if j is not None
                    else None
                )
                row = base | dict(
                    vertex_order=k,
                    x=p.x,
                    y=p.y,
                    next_x=q.x,
                    next_y=q.y,
                    neighbour="" if j is None else j,
                    edge_kind=edge.kind if edge else "box",
                )
                if edge and edge.origin:
                    row.update(ray_origin_x=edge.origin.x, ray_origin_y=edge.origin.y)
                if edge and edge.direction:
                    row.update(
                        direction_x=edge.direction.x, direction_y=edge.direction.y
                    )
                writer.writerow(row)


def write_json(diagram, path):
    def pair(p):
        return [p.x, p.y] if p is not None else None

    payload = dict(
        schema_version=1,
        bbox=diagram.box.as_tuple(),
        sites=[pair(p) for p in diagram.sites],
        statistics=diagram.statistics,
        cells=[
            dict(
                site=c.site,
                input_indices=c.input_indices,
                unbounded=c.unbounded,
                vertices=[pair(p) for p in c.vertices],
                boundary_neighbours=c.boundary_neighbours,
            )
            for c in diagram.cells
        ],
        edges=[
            dict(
                sites=e.sites,
                start=pair(e.start),
                end=pair(e.end),
                kind=e.kind,
                unbounded=e.unbounded,
                origin=pair(e.origin),
                direction=pair(e.direction),
            )
            for e in diagram.edges
        ],
        vertices=[
            dict(point=pair(v.point), defining=sorted(v.defining))
            for v in diagram.vertices
        ],
    )
    ensure_parent(path).write_text(
        json.dumps(payload, indent=2, allow_nan=False), encoding="utf-8"
    )

"""Bounded, on-demand replay of the actual Fortune event engine.

The completed clipped diagram is prepared for the final frame. A fresh sweep
then uses FortuneSweep.step(), also used by batch construction. Intermediate
frames never infer a beach line from the completed polygons. No frame history
is retained. Exactly collinear inputs show the production solver's special case.
"""

import math
from dataclasses import dataclass

from .diagram import construct
from .geometry import Box, Point, breakpoint_x, parabola_y
from .replay_state import clone_sweep
from .sweep import FortuneSweep, all_collinear, collinear_neighbours

MAX_REPLAY_SITES = 200


@dataclass(frozen=True, slots=True)
class ArcCurve:
    site: int
    points: tuple[Point, ...]
    limiting: bool = False


def stale(event):
    return event.kind == 1 and (
        not event.arc.alive or event.arc.generation != event.generation
    )


class SweepReplay:
    def __init__(self, coordinates, box=None, *, debug=False):
        coordinates = [
            p if isinstance(p, Point) else Point(float(p[0]), float(p[1]))
            for p in coordinates
        ]
        if len(set(coordinates)) > MAX_REPLAY_SITES:
            raise ValueError(
                f"Replay supports at most {MAX_REPLAY_SITES} distinct sites; "
                "use batch mode for larger inputs."
            )
        self.diagram = construct(coordinates, box, debug=debug)
        self.sites, self.box = self.diagram.sites, self.diagram.box
        self.debug = debug
        xmin, xmax = min(p.x for p in self.sites), max(p.x for p in self.sites)
        ymin, ymax = min(p.y for p in self.sites), max(p.y for p in self.sites)
        self.scale = max(xmax - xmin, ymax - ymin) or max(
            1.0, abs(xmin) * 1e-6, abs(ymin) * 1e-6
        )
        self.ox, self.oy = xmin + (xmax - xmin) / 2, ymin + (ymax - ymin) / 2
        self.points = [self.normalize(p) for p in self.sites]
        b = self.box
        self.view = Box(
            min(b.xmin, xmin),
            min(b.ymin, ymin),
            max(b.xmax, xmax),
            max(b.ymax, ymax),
        )
        span = max(self.view.xmax - self.view.xmin, self.view.ymax - self.view.ymin)
        self.top_y = self.view.ymax + 0.1 * span
        self.bottom_y = self.view.ymin - 0.1 * span
        self.special_case = all_collinear(self.sites)
        self._checkpoints = {}
        self._final_checkpoint = None
        self.reset()
        # Prepare the bounded seek index once, keeping pointer motion cheap.
        if not self.special_case:
            while not self.done:
                self.step_event()
            self.reset()

    def normalize(self, p):
        return Point((p.x - self.ox) / self.scale, (p.y - self.oy) / self.scale)

    def world(self, p):
        return Point(self.ox + self.scale * p.x, self.oy + self.scale * p.y)

    @property
    def done(self):
        return self.special_case or not self.sweep.queue

    @property
    def sweep_y(self):
        return self.oy + self.scale * self.display_y

    def label(self, site):
        return "/".join(map(str, self.diagram.cells[site].input_indices))

    def reset(self):
        self.sweep = (
            collinear_neighbours(self.points)
            if self.special_case
            else FortuneSweep(
                self.points, debug=self.debug, original_points=self.sites
            ).start()
        )
        self.processed = set(range(len(self.sites))) if self.special_case else set()
        self.display_y = (self.top_y - self.oy) / self.scale
        self.last_event = None
        self.last_stale = False
        self.event_count = 0
        self._subframe = 0
        self._segment_start = self.display_y
        self._fraction = 0.0

    def step_event(self):
        if self.done:
            return None
        previous_stale = self.sweep.statistics["stale_events"]
        event = self.sweep.step()
        self.display_y = self.sweep.sweep_y
        self.last_event = event
        self.last_stale = self.sweep.statistics["stale_events"] > previous_stale
        self.event_count += 1
        if event.kind == 0:
            self.processed.add(event.site)
        self._subframe = 0
        self._segment_start = self.display_y
        self._fraction = 0.0
        self._remember_checkpoint()
        return event

    def _remember_checkpoint(self):
        bucket = min(7, int(self.position * 8))
        if self.done:
            if self._final_checkpoint is None:
                self._final_checkpoint = self._checkpoint()
        elif bucket not in self._checkpoints:
            self._checkpoints[bucket] = self._checkpoint()

    def _checkpoint(self):
        engine, event = clone_sweep(self.sweep, self.last_event)
        return (
            engine,
            event,
            frozenset(self.processed),
            self.event_count,
            self.last_stale,
        )

    def _restore(self, checkpoint):
        engine, event, processed, count, last_stale = checkpoint
        self.sweep, self.last_event = clone_sweep(engine, event)
        self.processed = set(processed)
        self.event_count = count
        self.last_stale = last_stale
        self.display_y = self.sweep.sweep_y
        self._segment_start = self.display_y
        self._subframe = 0
        self._fraction = 0.0

    @property
    def position(self):
        if self.done:
            return 1.0
        return min(
            1.0, max(0.0, (self.top_y - self.sweep_y) / (self.top_y - self.bottom_y))
        )

    def seek_y(self, world_y):
        """Seek using real event dispatch; upward motion replays from the start.

        Up to eight height-bucket checkpoints and one final state keep storage
        linear and avoid reconstructing the entire sweep on every pointer move.
        """
        if not math.isfinite(world_y):
            raise ValueError("Sweep position must be finite")
        if self.special_case:
            return
        target = (world_y - self.oy) / self.scale
        candidates = [s for s in self._checkpoints.values() if s[0].sweep_y >= target]
        best = max(candidates, key=lambda s: s[3]) if candidates else None
        if target > self.display_y or (best and best[3] > self.event_count):
            if best:
                self._restore(best)
            else:
                self.reset()
        while self.sweep.queue and self.sweep.queue.peek().y >= target:
            self.step_event()
        self.display_y = target
        self._segment_start = target
        self._subframe = 1
        self._fraction = 0.0

    def seek(self, position):
        if not math.isfinite(position) or not 0 <= position <= 1:
            raise ValueError("Replay position must be between 0 and 1")
        if position == 0:
            self.reset()
        elif position == 1:
            if self._final_checkpoint:
                self._restore(self._final_checkpoint)
            else:
                while not self.done:
                    self.step_event()
        else:
            self.seek_y(self.top_y + position * (self.bottom_y - self.top_y))

    def advance(self, event_fraction):
        """Advance a wall-time budget measured in events, independent of FPS."""
        if not math.isfinite(event_fraction) or event_fraction < 0:
            raise ValueError("Playback advance must be finite and nonnegative")
        budget = self._fraction + event_fraction
        while not self.done:
            target = min(self.sweep.sweep_y, self.sweep.queue.peek().y)
            if target == self._segment_start:
                self.step_event()
                continue
            if budget < 1:
                self._fraction = budget
                self._subframe = int(budget > 0)
                self.display_y = (1 - budget) * self._segment_start + budget * target
                return
            budget -= 1
            self.step_event()

    def tick(self, substeps=8):
        """Move toward, never past, the next event before processing it.

        Same-height events remain distinct steps. Subdivision is per event,
        avoiding arbitrarily long waits for far-away circumcircle bottoms.
        """
        if substeps < 1:
            raise ValueError("Substeps must be positive")
        if self.done:
            return None
        target = min(self.sweep.sweep_y, self.sweep.queue.peek().y)
        self._subframe += 1
        if self._subframe >= substeps or target == self.display_y:
            return self.step_event()
        fraction = self._subframe / substeps
        self.display_y = (1 - fraction) * self._segment_start + fraction * target
        return None

    def curves(self, samples=65):
        """Sample only each live arc's actual breakpoint interval."""
        if samples < 2:
            raise ValueError("At least two arc samples are required")
        if self.done:
            return []
        lo, hi = (
            self.normalize(Point(self.view.xmin, self.view.ymin)),
            self.normalize(Point(self.view.xmax, self.view.ymax)),
        )
        pad = 0.06 * max(hi.x - lo.x, hi.y - lo.y)
        result = []
        for arc in self.sweep.beach:
            focus = self.points[arc.site]
            if focus.y == self.display_y:
                # The new parabola is a vertical limiting fragment at birth.
                heights = [
                    parabola_y(self.points[a.site], focus.x, self.display_y)
                    for a in (arc.prev, arc.next)
                    if a and self.points[a.site].y > self.display_y
                ]
                top = min(hi.y + pad, min(heights) if heights else hi.y + pad)
                result.append(
                    ArcCurve(
                        arc.site,
                        (
                            self.world(focus),
                            self.world(Point(focus.x, max(focus.y, top))),
                        ),
                        True,
                    )
                )
                continue
            if focus.y < self.display_y:
                raise ArithmeticError("Replay arc focus lies below the sweep")
            left = (
                breakpoint_x(self.points[arc.prev.site], focus, self.display_y)
                if arc.prev
                else lo.x - pad
            )
            right = (
                breakpoint_x(focus, self.points[arc.next.site], self.display_y)
                if arc.next
                else hi.x + pad
            )
            left, right = max(left, lo.x - pad), min(right, hi.x + pad)
            if left >= right:
                continue
            points = []
            for i in range(samples):
                x = left + (right - left) * i / (samples - 1)
                y = parabola_y(focus, x, self.display_y)
                p = self.world(Point(x, y))
                points.append(
                    p
                    if math.isfinite(p.x) and math.isfinite(p.y)
                    else Point(math.nan, math.nan)
                )
            result.append(ArcCurve(arc.site, tuple(points)))
        return result

    def breakpoints(self):
        result = []
        if self.done:
            return result
        for left in self.sweep.beach:
            if left.next is None:
                continue
            right = left.next
            a, b = self.points[left.site], self.points[right.site]
            x = breakpoint_x(a, b, self.display_y)
            focus = a if a.y > self.display_y else b
            if focus.y <= self.display_y:
                continue
            p = self.world(Point(x, parabola_y(focus, x, self.display_y)))
            if math.isfinite(p.x) and math.isfinite(p.y):
                result.append((tuple(sorted((left.site, right.site))), p))
        return result

    def update(self, model):
        revision = (id(model.sweep), model.event_count, model.done, model.last_stale)
        self.event_changed = revision != self.revision
        if self.event_changed:
            self._update_event(model)
            self.revision = revision
        self.cells.set_visible(model.done)
        self.height.set_text(
            "" if model.special_case else f"Sweep y = {model.sweep_y:.6g}"
        )
        if model.done:
            self.hide_geometry()
            self.cells.set_visible(True)
            return
        curves = model.curves()
        for limiting, collection in ((False, self.beach), (True, self.limiting)):
            collection.set_segments(
                [
                    [(p.x, p.y) for p in c.points]
                    for c in curves
                    if c.limiting == limiting
                ]
            )
            collection.set_visible(True)
        breakpoints = model.breakpoints()
        self.breakpoints.set_offsets(offsets(p for _, p in breakpoints))
        self.breakpoints.set_visible(True)
        active = {}
        for key, p in breakpoints:
            active.setdefault(key, []).append(p)
        lines = []
        for key, edge in model.sweep.edges.items():
            ends = [model.world(model.sweep.vertices[v].point) for v in edge.vertices]
            moving = active.get(key, [])
            if len(ends) >= 2:
                lines.append((ends[0], ends[1]))
            elif ends:
                lines.extend((ends[0], p) for p in moving)
            elif len(moving) >= 2:
                lines.append((moving[0], moving[1]))
            elif moving and model.points[key[0]].y == model.points[key[1]].y:
                lines.append((moving[0], Point(moving[0].x, self.ylim[1])))
        self.edges.set_segments([[(p.x, p.y), (q.x, q.y)] for p, q in lines])
        self.edges.set_visible(True)
        y = model.sweep_y
        self.sweep.set_ydata([y, y])
        self.sweep.set_visible(self.ylim[0] <= y <= self.ylim[1])
        self.outside.set_visible(not self.sweep.get_visible())
        if self.outside.get_visible():
            above = y > self.ylim[1]
            self.outside.set_text("Sweep above view" if above else "Sweep below view")
            self.outside.set_position((0.02, 0.97 if above else 0.02))
            self.outside.set_verticalalignment("top" if above else "bottom")
        candidate = model.sweep.queue.peek()
        if (
            model.last_event
            and model.last_event.kind == 1
            and not model.last_stale
            and model._subframe == 0

            candidate = model.last_event
        self.circle.set_visible(False)

            candidate

                centre = model.world(candidate.centre)
                self.circle.center = (centre.x, centre.y)
                self.circle.set_radius(radius)
                self.circle.set_visible(True)

                order += " | ..."
            self.order.set_text(

            )


class CachedPainter:
    """Two cached layers: static layout, then event text and control state."""

    def __init__(self, figure, event_artists, motion_artists, motion_regions=None):
        self.figure, self.canvas = figure, figure.canvas
        self.event_artists = event_artists
        self.motion_regions = motion_regions
        self.motion_artists = sorted(motion_artists, key=lambda a: a.get_zorder())
        self.artists = list(dict.fromkeys(event_artists + motion_artists))
        self.enabled = bool(self.canvas.supports_blit)
        self.base = self.event_background = None
        self.event_dirty = True
        for artist in self.artists:
            artist.set_animated(self.enabled)
        self.cid = self.canvas.mpl_connect("draw_event", self._on_draw)

    def _draw(self, artists):
        for artist in artists:
            if artist.get_visible():
                self.figure.draw_artist(artist)

    def _events(self):
        self.canvas.restore_region(self.base)
        self._draw(self.event_artists)
        self.event_background = self.canvas.copy_from_bbox(self.figure.bbox)
        self.event_dirty = False

    def _on_draw(self, event):
        if not self.enabled or self.canvas.is_saving():
            return
        self.base = self.canvas.copy_from_bbox(self.figure.bbox)
        self._events()
        self._draw(self.motion_artists)
        self.canvas.blit(self.figure.bbox)

    def paint(self, *, event_dirty=False):
        if not self.enabled:
            self.canvas.draw_idle()
            return
        self.event_dirty |= event_dirty
        if self.base is None:
            self.canvas.draw()
            return
        full_blit = self.event_dirty
        if self.event_dirty:
            self._events()
        self.canvas.restore_region(self.event_background)
        self._draw(self.motion_artists)
        if full_blit or self.motion_regions is None:
            self.canvas.blit(self.figure.bbox)
        else:
            for region in self.motion_regions():
                self.canvas.blit(region)

    def invalidate(self):
        self.base = self.event_background = None

    @contextmanager
    def full_render(self):
        states = [a.get_animated() for a in self.artists]
        try:
            for a in self.artists:
                a.set_animated(False)
            yield
        finally:
            for a, state in zip(self.artists, states):
                a.set_animated(state)
            self.invalidate()

    def close(self):
        self.canvas.mpl_disconnect(self.cid)

    if (left[1]===right[1]) return left[0]+(right[0]-left[0])/2;
    const dp=left[1]-d,dq=right[1]-d;
    if (dp<=0) return left[0];
    if (dq<=0) return right[0];
    const dx=right[0]-left[0],a=right[1]-left[1],b=2*dp*dx;
    const c=-dp*dx*dx-a*dp*dq;
    let discriminant=b*b-4*a*c;
    if (discriminant<0) {
      if (discriminant < -2e-14*(b*b+Math.abs(4*a*c))) throw new Error("Breakpoint precision limit");
      discriminant=0;
    }
    const q=-0.5*(b+((b<0||Object.is(b,-0))?-1:1)*Math.sqrt(discriminant));
    const r1=q?q/a:-b/(2*a),r2=q?c/q:r1;

"""Export an offline, pointer-following Canvas replay of real Python events.

Fortune construction stays in Python. The browser receives immutable event
states and draws their parabolas; pointer movement makes no Python calls.
This bounded educational replay may store O(n^2) arc/queue snapshot data at
the 200-site cap. Normal construction and its benchmarks do not collect it.
"""

import heapq
import json
import webbrowser
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone
from pathlib import Path

from .geometry import Point
from .replay import SweepReplay, stale


def build_payload(coordinates, box=None, *, labels=True):
    model = SweepReplay(coordinates, box)
    engine = model.sweep
    events = {}
    edges = {}
    states = []

    def capture(step, last=None, discarded=False):
        for entry in engine.queue._heap:
            serial, event = entry[3], entry[-1]
            if serial not in events:
                events[serial] = dict(
                    kind=event.kind,
                    y=event.y,
                    x=event.x,
                    site=event.site,
                    defining=event.defining,
                    centre=[event.centre.x, event.centre.y] if event.centre else None,
                )
        for key, edge in engine.edges.items():
            record = edges.setdefault(key, dict(sites=list(key), created=step, ends=[]))
            for vertex in edge.vertices:
                if not any(end[0] == vertex for end in record["ends"]):
                    record["ends"].append([vertex, step])
        states.append(
            dict(
                y=engine.sweep_y if step else model.normalize(Point(0, model.top_y)).y,
                arcs=[arc.site for arc in engine.beach],
                queue=[
                    [e[3], stale(e[-1])] for e in heapq.nsmallest(8, engine.queue._heap)
                ],
                queued=len(engine.queue),
                last=last,
                discarded=discarded,
                counts=[
                    engine.statistics["site_events"],
                    engine.statistics["circle_events"],
                    engine.statistics["stale_events"],
                ],
                done=not bool(engine.queue),
            )
        )

    capture(0)
    step = 0
    while engine.queue:
        serial = engine.queue._heap[0][3]
        before = engine.statistics["stale_events"]
        model.step_event()
        step += 1
        capture(step, serial, engine.statistics["stale_events"] > before)
    view = model.view
    span = max(view.xmax - view.xmin, view.ymax - view.ymin)
    left = model.normalize(Point(view.xmin - 0.1 * span, view.ymin - 0.1 * span))
    right = model.normalize(Point(view.xmax + 0.1 * span, view.ymax + 0.1 * span))
    b = model.box
    low = model.normalize(Point(b.xmin, b.ymin))
    high = model.normalize(Point(b.xmax, b.ymax))
    payload = dict(
        generated_at=datetime.now(timezone(timedelta(hours=8))).isoformat(),
        source="Python FortuneSweep.step; no browser Voronoi implementation",
        sites=[[p.x, p.y, model.label(i)] for i, p in enumerate(model.points)],
        vertices=[[v.point.x, v.point.y] for v in engine.vertices],
        edges=list(edges.values()),
        events={str(k): v for k, v in events.items()},
        states=states,
        cells=[
            [[model.normalize(p).x, model.normalize(p).y] for p in c.vertices]
            for c in model.diagram.cells
        ],
        view=[left.x, left.y, right.x, right.y],
        box=[low.x, low.y, high.x, high.y],
        world=[model.ox, model.oy, model.scale],
        labels=labels,
        special_case=model.special_case,
    )
    return payload, model


def write_replay(coordinates, box=None, output="replay.html", *, labels=True):
    payload, model = build_payload(coordinates, box, labels=labels)
    folder = Path(__file__).parent
    template = (folder / "replay_canvas.html").read_text(encoding="utf-8")
    script = (folder / "replay_canvas.js").read_text(encoding="utf-8")
    data = json.dumps(
        payload, ensure_ascii=False, allow_nan=False, separators=(",", ":")
    )
    data = data.replace("<", "\\u003c").replace(">", "\\u003e").replace("&", "\\u0026")
    html = template.replace("__FORTUNE_DATA__", data).replace(
        "__FORTUNE_SCRIPT__", script
    )
    path = Path(output).with_suffix(".html")
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(html, encoding="utf-8")
    return CanvasReplay(path.resolve(), model)

class CanvasReplay:
    path: Path
    model: SweepReplay

    if open_browser:

"""Copy the small mutable event state for at most eight replay checkpoints.

Copying explicitly preserves live arc identities, heap serial priorities and
stale-event tombstones without following arbitrarily long neighbour cycles.
The construction algorithm does not use checkpoints.
"""

import itertools

from .beachline import Arc
from .events import Event
from .sweep import FortuneSweep, TraceEdge, Vertex


def clone_sweep(source, extra_event=None):
    result = FortuneSweep(
        source.points, debug=source.debug, original_points=source.original_points
    )
    arcs = {
        a: Arc(a.site, height=a.height, generation=a.generation, alive=a.alive)
        for a in source.beach
    }
    for old, new in list(arcs.items()):
        for link in ("left", "right", "parent", "prev", "next"):
            setattr(new, link, arcs.get(getattr(old, link)))
    result.beach.root = arcs.get(source.beach.root)
    result.beach.first = arcs.get(source.beach.first)
    result.beach.count = source.beach.count
    result.beach.max_height = source.beach.max_height

    def clone_event(event):
        if event is None:
            return None
        arc = event.arc
        if arc is not None and arc not in arcs:
            # A removed arc is only consulted by the stale-event guard.
            arcs[arc] = Arc(arc.site, generation=arc.generation, alive=arc.alive)
        return Event(
            event.y,
            event.x,
            event.kind,
            site=event.site,
            arc=arcs.get(arc),
            generation=event.generation,
            centre=event.centre,
            defining=event.defining,
        )

    result.queue._heap = [
        (*entry[:4], clone_event(entry[-1])) for entry in source.queue._heap
    ]
    # Each push consumes one serial; the next value equals the push count.
    result.queue._serial = itertools.count(source.queue.pushed)
    result.queue.peak_size = source.queue.peak_size
    result.queue.pushed = source.queue.pushed
    result.sweep_y = source.sweep_y
    result.neighbours = [set(ns) for ns in source.neighbours]
    result.edges = {
        key: TraceEdge(edge.sites, list(edge.vertices))
        for key, edge in source.edges.items()
    }
    result.vertices = [Vertex(v.point, set(v.defining)) for v in source.vertices]
    result._vertex_index = dict(source._vertex_index)
    result.statistics = dict(source.statistics)
    result._started = source._started
    if result.debug:
        result.beach.validate()
    return result, clone_event(extra_event)

"""Responsive playback, direct sweep dragging and position scrubbing."""

from pathlib import Path
from time import perf_counter

from .io import ensure_parent

from .replay_artists import CachedPainter, ReplayArtists, event_name

PLOT_RECT = (0.07, 0.30, 0.58, 0.60)
PANEL_RECT = (0.71, 0.31, 0.27, 0.58)


def draw_replay(model, axes, panel, *, labels=True):
    """Update persistent artists; also usable by headless frame/GIF exports."""
    scene = getattr(axes, "_fortune_replay_artists", None)
    if scene is None or scene.panel is not panel or scene.diagram is not model.diagram:
        scene = ReplayArtists(model, axes, panel, labels=labels)
        axes._fortune_replay_artists = scene
    scene.update(model)
    return scene


class ReplayController:
    def __init__(
        self,
        figure,
        axes,
        panel,
        model,
        *,
        output="replay.png",
        labels=True,
        clock=None,
    ):
        from matplotlib.widgets import Button, Slider

        self.figure, self.axes, self.panel, self.model = figure, axes, panel, model
        self.output, self.labels = Path(output), labels
        self.playing = self.closed = self._dragging = False
        self._clock = clock or perf_counter
        self._last_tick = self._clock()
        self._scene = self._painter = None
        self._paint_revision = None
        self.timer = figure.canvas.new_timer(interval=16)
        self.timer.add_callback(self.tick)
        self.status = ui_text(figure, 0.07, 0.025, "", fontsize=9, color="#344054")
        self.buttons = []
        for x, width, label, callback in (
            (0.07, 0.12, "Play", self.toggle_play),
            (0.21, 0.12, "Next event", self.next_event),
            (0.35, 0.12, "Reset", self.reset),
            (0.49, 0.14, "Save frame", self.save_frame),
        ):
            button = Button(
                figure.add_axes([x, 0.075, width, 0.045]), label, useblit=False
            )
            button.label = replace_widget_text(button.ax, button.label)
            button.drawon = False
            button.hovercolor = button.color
            button.on_clicked(callback)
            self.buttons.append(button)
        self.speed = Slider(
            figure.add_axes([0.76, 0.09, 0.18, 0.02]),
            "Events/s",
            0.5,
            30,
            valinit=6,
            valstep=0.5,
        )
        self.speed.label = replace_widget_text(self.speed.ax, self.speed.label)
        self.speed.valtext = replace_widget_text(self.speed.ax, self.speed.valtext)
        self.speed.drawon = False
        self.speed.on_changed(self.speed_changed)
        self.position = Slider(
            figure.add_axes([0.14, 0.15, 0.48, 0.022]),
            "Position",
            0,
            100,
            valinit=100 * model.position,
            valfmt="%1.0f%%",
        )
        self.position.label = replace_widget_text(self.position.ax, self.position.label)
        self.position.valtext = replace_widget_text(
            self.position.ax, self.position.valtext
        )
        self.position.drawon = False
        self.position.set_active(not model.special_case)
        self.position.on_changed(self.scrub)
        for name, callback in (
            ("key_press_event", self.key),
            ("close_event", self.close),
            ("button_press_event", self.press),
            ("motion_notify_event", self.motion),
            ("button_release_event", self.release),
        ):
            self.figure.canvas.mpl_connect(name, callback)
        self.redraw()

    def redraw(self, message=None, *, controls_changed=False):
        if self.closed:
            return
        try:
            self._scene = draw_replay(
                self.model, self.axes, self.panel, labels=self.labels
            )
        except (ValueError, ArithmeticError) as exc:
            self.pause()
            if self._scene:
                self._scene.hide_geometry()
            message = f"Replay drawing stopped: {exc}"
        if message is None:
            if self.model.special_case:
                message = "Collinear input: direct-bisector construction shown. Choose non-collinear sites to replay the sweep."
            elif self.model.done:
                message = f"Complete: {self.model.event_count} entries processed. Drag Position backward or Play to replay."
            elif self.model.last_event:
                message = f"Event {self.model.event_count}: {event_name(self.model, self.model.last_event)}"
                if self.model.last_stale:
                    message += " | stale prediction discarded"
                message += " | Drag the plot or Position; Space: play/pause; Right: next event."
            else:
                message = "Play or drag the plot / Position slider. Right: next event; R: reset."
        dirty = controls_changed or message != self.status.get_text()
        self.status.set_text(message)
        label = "Pause" if self.playing else "Play"
        dirty |= label != self.buttons[0].label.get_text()
        self.buttons[0].label.set_text(label)
        events = self.position.eventson
        self.position.eventson = False
        self.position.set_val(100 * self.model.position)
        self.position.eventson = events
        if self._scene is None:
            self.figure.canvas.draw_idle()
            return
        if self._painter is None:
            self._painter = CachedPainter(
                self.figure,
                self._scene.event_artists
                + [self.status, self.speed.ax]
                + [b.ax for b in self.buttons],
                self._scene.motion_artists + [self.position.ax],
                motion_regions=self.movement_regions,
            )
        dirty |= self._paint_revision != self._scene.revision
        self._painter.paint(event_dirty=dirty)
        self._paint_revision = self._scene.revision

    def movement_regions(self):
        from matplotlib.transforms import Bbox

        # Avoid copying the full Tk PhotoImage when only these regions move.
        scale = self.figure.dpi / 100
        slider = self.position.ax.bbox
        panel = self.panel.bbox
        baseline = panel.y0 + 0.055 * panel.height
        return (
            self.axes.bbox.padded(4 * scale),
            Bbox.from_extents(
                slider.x0 - 100 * scale,
                slider.y0 - 15 * scale,
                slider.x1 + 70 * scale,
                slider.y1 + 20 * scale,
            ),
            Bbox.from_extents(
                panel.x0 - 3 * scale,
                baseline - 10 * scale,
                panel.x1 + 3 * scale,
                baseline + 22 * scale,
            ),
        )

    def pause(self):
        self.playing = False
        self.timer.stop()

    def toggle_play(self, event=None):
        if self.model.special_case or self.closed:
            return
        if self.playing:
            self.pause()
        else:
            if self.model.done:
                self.model.reset()
            self.playing = True
            self._last_tick = self._clock()
            self.timer.start()
        self.redraw()

    def next_event(self, event=None):
        self.pause()
        try:
            self.model.step_event()
            self.redraw()
        except (ValueError, ArithmeticError) as exc:
            self.redraw(f"Replay stopped: {exc}")

    def tick(self):
        # Returning False would remove this callback from Matplotlib's timer.
        if not self.playing or self.closed:
            return
        now = self._clock()
        elapsed = min(0.25, max(0.0, now - self._last_tick))
        self._last_tick = now
        frame_started = perf_counter()
        try:
            self.model.advance(elapsed * self.speed.val)
            if self.model.done:
                self.pause()
            self.redraw()
        except (ValueError, ArithmeticError) as exc:
            self.pause()
            self.redraw(f"Replay stopped: {exc}")
        # Tk schedules its next callback after this one returns. Subtract the
        # drawing time instead of adding a fixed 16 ms delay to every frame.
        self.timer.interval = max(
            1, round(1000 / 60 - (perf_counter() - frame_started) * 1000)
        )

    def speed_changed(self, value):
        self._last_tick = self._clock()
        self.redraw(controls_changed=True)

    def scrub(self, value):
        self.pause()
        try:
            self.model.seek(float(value) / 100)
            self.redraw()
        except (ValueError, ArithmeticError) as exc:
            self.redraw(f"Replay stopped: {exc}")

    def _drag_y(self, value):
        self.pause()
        try:
            self.model.seek_y(min(self.model.top_y, max(self.model.bottom_y, value)))
            self.redraw()
        except (ValueError, ArithmeticError) as exc:
            self.redraw(f"Replay stopped: {exc}")

    def press(self, event):
        if (
            event.inaxes is self.axes
            and event.button == 1
            and event.ydata is not None
            and not self.model.special_case
            and self.axes.get_navigate_mode() is None
        ):
            self._dragging = True
            self._drag_y(event.ydata)

    def motion(self, event):
        if self._dragging and event.inaxes is self.axes and event.ydata is not None:
            self._drag_y(event.ydata)

    def release(self, event):
        if self._dragging and event.inaxes is self.axes and event.ydata is not None:
            self._drag_y(event.ydata)
        self._dragging = False

    def reset(self, event=None):
        self.pause()
        self.model.reset()
        self.redraw()

    def save_frame(self, event=None):
        try:
            if self._painter:
                with self._painter.full_render():
                    self.figure.savefig(ensure_parent(self.output), dpi=160)
            else:
                self.figure.savefig(ensure_parent(self.output), dpi=160)
            self.redraw(f"Saved current replay frame: {self.output.name}")
        except (OSError, ValueError) as exc:
            self.redraw(f"Could not save frame: {exc}")

    def key(self, event):
        if event.key in (" ", "space"):
            self.toggle_play()
        elif event.key == "right":
            self.next_event()
        elif event.key == "r":
            self.reset()

    def close(self, event=None):
        self.pause()
        self.closed = True
        if self._painter:
            self._painter.close()

"""Fortune's event-driven sweep, producing adjacency and traced Voronoi edges.

No cell is constructed here using comparisons with every other site. The
adjacency graph is generated only by site splits and disappearing arcs.
"""

import math
from dataclasses import dataclass, field

from .beachline import Arc, BeachLine
from .circle import candidate_circle
from .events import Event, EventQueue
from .geometry import Point, circle_identity, orient


@dataclass(slots=True)
class Vertex:
    point: Point
    defining: set[int]


@dataclass(slots=True)
class TraceEdge:
    sites: tuple[int, int]
    vertices: list[int] = field(default_factory=list)


class FortuneSweep:
    def __init__(self, points, debug=False, original_points=None):
        self.points = points
        self.original_points = (
            original_points if original_points is not None else points
        )
        self.debug = debug
        self.beach = BeachLine()
        self.queue = EventQueue()
        self.sweep_y = math.inf
        self.neighbours = [set() for _ in points]
        self.edges = {}
        self.vertices = []
        self._vertex_index = {}
        self.statistics = dict(site_events=0, circle_events=0, stale_events=0)
        self._started = False

    def edge(self, a, b, vertex=None):
        if a == b:
            raise ArithmeticError("An edge cannot separate a site from itself")
        key = (min(a, b), max(a, b))
        if key not in self.edges:
            self.edges[key] = TraceEdge(key)
            self.neighbours[a].add(b)
            self.neighbours[b].add(a)
        edge = self.edges[key]
        if vertex is not None and vertex not in edge.vertices:
            edge.vertices.append(vertex)
        return edge

    def schedule(self, arc):
        if arc is None or not arc.alive:
            return
        arc.generation += 1
        if arc.prev is None or arc.next is None:
            return
        ids = (arc.prev.site, arc.site, arc.next.site)
        if orient(*(self.original_points[i] for i in ids)) >= 0:
            return
        result = candidate_circle(*(self.points[i] for i in ids), self.sweep_y)
        if result:
            centre, bottom = result
            self.queue.push(
                Event(
                    bottom,
                    centre.x,
                    1,
                    arc=arc,
                    generation=arc.generation,
                    centre=centre,
                    defining=ids,
                )
            )

    def site_event(self, site):
        self.statistics["site_events"] += 1
        if self.beach.root is None:
            self.beach.insert_after(None, Arc(site))
            return
        point = self.points[site]
        old = self.beach.locate(point.x, self.sweep_y, self.points)
        before, after = old.prev, old.next
        old.generation += 1
        if self.points[old.site].y == point.y:
            # A whole equal-height batch has vertical limiting parabolas. It
            # inserts an adjacent arc instead of splitting a finite parabola.
            new = Arc(site)
            self.beach.insert_after(
                old if point.x >= self.points[old.site].x else old.prev, new
            )
            self.edge(old.site, site)
            for arc in (before, old, new, after):
                self.schedule(arc)
            return
        new, right = Arc(site), Arc(old.site)
        self.beach.insert_after(old, new)
        self.beach.insert_after(new, right)
        self.edge(old.site, site)
        for arc in (before, old, new, right, after):
            self.schedule(arc)

    def circle_event(self, event):
        middle = event.arc
        if not middle.alive or middle.generation != event.generation:
            self.statistics["stale_events"] += 1
            return
        left, right = middle.prev, middle.next
        if (
            left is None
            or right is None
            or (left.site, middle.site, right.site) != event.defining
        ):
            raise ArithmeticError("A valid circle event lost its neighbourhood")
        key = circle_identity(*(self.original_points[i] for i in event.defining))
        if key is None:
            self.statistics["stale_events"] += 1
            return
        vertex = self._vertex_index.get(key)
        if vertex is None:
            vertex = len(self.vertices)
            self.vertices.append(Vertex(event.centre, set(event.defining)))
            self._vertex_index[key] = vertex
        else:
            self.vertices[vertex].defining.update(event.defining)
        self.edge(left.site, middle.site, vertex)
        self.edge(middle.site, right.site, vertex)
        self.edge(left.site, right.site, vertex)
        self.beach.remove(middle)
        self.statistics["circle_events"] += 1
        self.schedule(left)
        self.schedule(right)

    def start(self):
        """Queue sites once; replay and batch execution share this entry point."""
        if not self._started:
            for i, p in enumerate(self.points):
                self.queue.push(Event(p.y, p.x, 0, site=i))
            self._started = True
        return self

    def step(self):
        """Process exactly one heap entry, including an invalidated prediction.

        Return the processed event, or None after completion. The same handlers
        and generation guard are used by ordinary construction and replay.
        """
        self.start()
        if not self.queue:
            self._finish_statistics()
            return None
        event = self.queue.pop()
        self.sweep_y = min(self.sweep_y, event.y)
        if event.kind == 0:
            self.site_event(event.site)
        else:
            self.circle_event(event)
        if self.debug:
            self.beach.validate()
        if not self.queue:
            self._finish_statistics()
        return event

    def _finish_statistics(self):
        self.statistics.update(
            queue_peak=self.queue.peak_size,
            events_pushed=self.queue.pushed,
            beach_max_height=self.beach.max_height,
            trace_edges=len(self.edges),
        )

    def run(self):
        while self.step() is not None:
            pass
        return self


def collinear_neighbours(points):
    """One-dimensional degeneration: consecutive sites alone share bisectors."""
    sweep = FortuneSweep(points)
    order = sorted(range(len(points)), key=lambda i: (points[i].x, points[i].y))
    for a, b in zip(order, order[1:]):
        sweep.edge(a, b)
    sweep.statistics["collinear_special_case"] = True
    return sweep


def all_collinear(points):
    if len(points) < 3:
        return True
    a, b = min(points, key=lambda p: (p.x, p.y)), max(points, key=lambda p: (p.x, p.y))
    return all(orient(a, b, p) == 0 for p in points)

"""Cached bitmap text for the live Agg/Tk UI, with normal vector export.

Matplotlib's glyph-by-glyph text drawing dominated the replay profile. Pillow
renders these plain interface strings once; subsequent draws composite a small
RGBA bitmap. Font and string caches are bounded. Mathematical/report text is
unaffected, and SVG/PDF backends retain Matplotlib's regular text renderer.
"""

from functools import lru_cache

import numpy as np
from matplotlib import colors, font_manager
from matplotlib.backends.backend_agg import RendererAgg
from matplotlib.text import Text
from PIL import Image, ImageDraw, ImageFont


@lru_cache(maxsize=16)
def _font(path, size):
    return ImageFont.truetype(path, size, layout_engine=ImageFont.Layout.BASIC)


@lru_cache(maxsize=256)
def _bitmap(text, path, size, rgba, spacing):
    font = _font(path, size)
    measure = ImageDraw.Draw(Image.new("RGBA", (1, 1)))
    box = measure.multiline_textbbox(
        (0, 0), text, font=font, anchor="ls", spacing=spacing
    )
    left, top, right, bottom = box
    pad = 1
    image = Image.new(
        "RGBA", (max(1, right - left + 2 * pad), max(1, bottom - top + 2 * pad))
    )
    ImageDraw.Draw(image).multiline_text(
        (pad - left, pad - top),
        text,
        font=font,
        fill=rgba,
        anchor="ls",
        spacing=spacing,
    )
    return np.ascontiguousarray(np.asarray(image)[::-1]), bottom + pad


class RasterText(Text):
    def draw(self, renderer):
        text = self.get_text()
        if not self.get_visible() or not text:
            self.stale = False
            return
        if (
            not isinstance(renderer, RendererAgg)
            or self.get_rotation() != 0
            or "$" in text
            or self.get_bbox_patch() is not None
        ):
            return super().draw(renderer)
        size = max(1, round(renderer.points_to_pixels(self.get_fontsize())))
        path = font_manager.findfont(self.get_fontproperties())
        rgba = tuple(
            round(c * 255) for c in colors.to_rgba(self.get_color(), self.get_alpha())
        )
        linespacing = self.get_linespacing()
        # Matplotlib 3.11 can return "auto" for the default spacing.
        linespacing = linespacing if isinstance(linespacing, (int, float)) else 1.2
        spacing = max(0, round(size * (linespacing - 1)))
        pixels, baseline = _bitmap(text, path, size, rgba, spacing)
        x, y = self.get_transform().transform(self.get_position())
        height, width = pixels.shape[:2]
        ha, va = self.get_horizontalalignment(), self.get_verticalalignment()
        if ha == "center":
            x -= width / 2
        elif ha == "right":
            x -= width
        if va == "top":
            y -= height
        elif va in ("center", "center_baseline"):
            y -= height / 2
        elif va == "baseline":
            y -= baseline
        gc = renderer.new_gc()
        try:
            if self.get_clip_on():
                gc.set_clip_rectangle(self.get_clip_box())
            renderer.draw_image(gc, round(x), round(y), pixels)
        finally:
            gc.restore()
        self.stale = False

    if labels:
        for c in diagram.cells:

import math
import random

from fortune.beachline import Arc, BeachLine
from fortune.geometry import Point


@pytest.mark.parametrize("seed", range(5))
def test_avl_random_insert_and_delete(seed):
    rng, tree, expected = random.Random(seed), BeachLine(), []
    for i in range(600):
        index = rng.randrange(len(expected) + 1)
        new = Arc(i)
        tree.insert_after(expected[index - 1] if index else None, new)
        expected.insert(index, new)
        tree.validate()
        assert list(tree) == expected
    assert tree.root.height < 1.45 * math.log2(len(expected) + 2) + 1
    rng.shuffle(expected)
    for node in expected:
        tree.remove(node)
        tree.validate()
    assert tree.root is None and tree.count == 0


def test_ordered_locate_for_equal_height_sites():
    tree = BeachLine()
    points = [Point(-2, 1), Point(0, 1), Point(2, 1)]
    previous = None
    for i in range(3):
        new = Arc(i)
        tree.insert_after(previous, new)
        previous = new
    assert tree.locate(-10, 0, points).site == 0
    assert tree.locate(0, 0, points).site == 1
    assert tree.locate(10, 0, points).site == 2


def test_removed_arc_cannot_be_deleted_again():
    tree, node = BeachLine(), Arc(0)
    tree.insert_after(None, node)
    tree.remove(node)
    with pytest.raises(ValueError):
        tree.remove(node)

from fortune.beachline import Arc, BeachLine
from fortune.events import Event, EventQueue
from fortune.geometry import Point
from fortune.sweep import FortuneSweep


def _topology(sweep):
    """Freeze topology, including arc identities and mutable edge contents."""
    return (
        sweep.beach.root,
        sweep.beach.first,
        sweep.beach.count,
        tuple(
            (
                arc,
                arc.site,
                arc.left,
                arc.right,
                arc.parent,
                arc.prev,
                arc.next,
                arc.height,
                arc.generation,
                arc.alive,
            )
            for arc in sweep.beach
        ),
        tuple(frozenset(group) for group in sweep.neighbours),
        tuple(
            sorted(
                (key, edge.sites, tuple(edge.vertices))
                for key, edge in sweep.edges.items()
            )
        ),
        tuple((vertex.point, frozenset(vertex.defining)) for vertex in sweep.vertices),
        dict(sweep._vertex_index),
    )


def _dispatch(sweep, event):
    sweep.sweep_y = min(sweep.sweep_y, event.y)
    if event.kind == 0:
        sweep.site_event(event.site)
    else:
        sweep.circle_event(event)
    sweep.beach.validate()


def _pending_triangle():
    # The clockwise triple (-2, 0), (0, 2), (2, 0) lies on
    # the circle centred at (0, 0), with bottom y = -2.
    sweep = FortuneSweep(
        [Point(0, 2), Point(-2, 0), Point(2, 0)],
        debug=True,
    )
    for site in range(3):
        sweep.sweep_y = sweep.points[site].y
        sweep.site_event(site)
        sweep.beach.validate()

    event = sweep.queue.pop()
    assert not sweep.queue
    assert event.defining == (1, 0, 2)
    assert event.centre == Point(0, 0)
    assert event.y == -2
    assert event.arc.alive
    return sweep, event


def test_site_insertion_supersedes_live_prediction_without_topology_changes():
    points = [
        Point(0, 2),
        Point(-2, 0),
        Point(2, 0),
        Point(-4, -1),
    ]
    sweep = FortuneSweep(points, debug=True)
    for site, point in enumerate(points):
        sweep.queue.push(Event(point.y, point.x, 0, site=site))

    # Process the initial three sites through the real event queue.
    for expected_site in range(3):
        event = sweep.queue.pop()
        assert event.kind == 0 and event.site == expected_site
        _dispatch(sweep, event)

    # Capture the prediction without removing it from the logical queue.
    site_event = sweep.queue.pop()
    assert site_event.kind == 0 and site_event.site == 3
    pending = sweep.queue.pop()
    assert not sweep.queue
    assert pending.defining == (1, 0, 2)
    assert pending.centre == Point(0, 0)
    assert pending.y == -2
    target = pending.arc
    old_predecessor = target.prev
    sweep.queue.push(pending)

    _dispatch(sweep, site_event)

    # The new site splits the left neighbour. The target survives, but its
    # predecessor is now a different Arc carrying the same site ID.
    assert target.alive
    assert target.prev is not old_predecessor
    assert target.prev.site == old_predecessor.site == 1
    assert (target.prev.site, target.site, target.next.site) == pending.defining
    assert target.generation != pending.generation
    assert sweep.statistics["site_events"] == 4

    # Dispatch any earlier replacement predictions normally. The old
    # prediction must still be harmless when its actual heap turn arrives.
    while True:
        assert sweep.queue
        event = sweep.queue.pop()
        if event is pending:
            break
        _dispatch(sweep, event)

    assert target.alive
    assert (target.prev.site, target.site, target.next.site) == pending.defining
    before = _topology(sweep)
    statistics = sweep.statistics.copy()
    _dispatch(sweep, event)

    assert _topology(sweep) == before
    assert sweep.statistics == {
        **statistics,
        "stale_events": statistics["stale_events"] + 1,
    }


def test_duplicate_circle_delivery_after_arc_removal_is_harmless():
    sweep, prediction = _pending_triangle()
    target = prediction.arc

    # Deliberately duplicate delivery of a genuine scheduled prediction.
    sweep.queue.push(prediction)
    sweep.queue.push(prediction)
    _dispatch(sweep, sweep.queue.pop())

    assert not target.alive
    assert all(arc is not target for arc in sweep.beach)
    assert sweep.statistics["circle_events"] == 1
    assert len(sweep.vertices) == 1
    assert sweep.vertices[0].point == Point(0, 0)
    assert sweep.vertices[0].defining == {0, 1, 2}
    assert set(sweep.edges) == {(0, 1), (0, 2), (1, 2)}

    before = _topology(sweep)
    statistics = sweep.statistics.copy()
    delayed = sweep.queue.pop()
    assert delayed is prediction
    _dispatch(sweep, delayed)

    assert not sweep.queue
    assert _topology(sweep) == before
    assert sweep.statistics == {
        **statistics,
        "stale_events": statistics["stale_events"] + 1,
    }


def test_two_child_deletion_preserves_queued_deep_successor_identity():
    beach = BeachLine()
    arcs = [Arc(site) for site in range(16)]
    previous = None
    for arc in arcs[:15]:
        beach.insert_after(previous, arc)
        previous = arc
    beach.insert_after(arcs[8], arcs[15])
    beach.validate()

    victim = arcs[7]
    successor = arcs[8]
    assert victim.left is not None and victim.right is not None
    assert successor is victim.next
    assert successor.parent is not victim
    assert successor.left is None
    assert successor.right is arcs[15]

    queue = EventQueue()
    prediction = Event(
        -10,
        0,
        1,
        arc=successor,
        generation=successor.generation,
        centre=Point(0, 0),
        defining=(7, 8, 15),
    )
    queue.push(prediction)
    survivors = [arc for arc in beach if arc is not victim]
    payloads = [(arc, arc.site, arc.generation) for arc in survivors]

    beach.remove(victim)
    beach.validate()

    assert not victim.alive
    assert list(beach) == survivors
    for arc, site, generation in payloads:
        assert arc.alive
        assert arc.site == site
        assert arc.generation == generation

    delivered = queue.pop()
    assert delivered is prediction
    assert delivered.arc is successor
    assert delivered.arc.alive
    assert delivered.arc.site == 8
    assert delivered.arc.generation == delivered.generation
    assert successor.prev is arcs[6]
    assert successor.next is arcs[15]

    # Use the retained handle, rather than searching for its site again.
    # The old neighbourhood prediction is not dispatched as a valid circle.
    beach.remove(delivered.arc)
    beach.validate()
    assert list(beach) == [arc for arc in survivors if arc is not successor]
    assert arcs[6].next is arcs[15]
    assert arcs[15].prev is arcs[6]


def test_exact_heap_priority_ties_are_fifo_across_interleaved_pushes():
    queue = EventQueue()
    circle_a = Event(5, 0, 1, arc=Arc(10))
    circle_b = Event(5, 0, 1, arc=Arc(11))
    site_a = Event(5, 0, 0, site=20)
    site_b = Event(5, 0, 0, site=21)
    higher = Event(6, 100, 1, arc=Arc(12))
    lower = Event(4, -100, 0, site=22)
    left_circle = Event(5, -1, 1, arc=Arc(13))

    for event in (
        circle_a,
        lower,
        site_a,
        circle_b,
        higher,
        left_circle,
        site_b,
    ):
        queue.push(event)

    assert queue.pop() is higher
    assert queue.pop() is site_a

    # New ties must follow older pending ties, even after partial draining.
    site_c = Event(5, 0, 0, site=23)
    circle_c = Event(5, 0, 1, arc=Arc(14))
    queue.push(circle_c)
    queue.push(site_c)

    expected = [
        site_b,
        site_c,
        left_circle,
        circle_a,
        circle_b,
        circle_c,
        lower,
    ]
    for event in expected:
        assert queue.pop() is event
    assert not queue

def signature(sweep):
    return (
        tuple((v.point, frozenset(v.defining)) for v in sweep.vertices),
        tuple((key, tuple(e.vertices)) for key, e in sorted(sweep.edges.items())),
        tuple(frozenset(ns) for ns in sweep.neighbours),
        dict(sweep.statistics),
    )


def test_queue_preview_is_ordered_without_popping():
    queue = EventQueue()
    events = [Event(2, 0, 1), Event(2, 1, 0), Event(3, 0, 0), Event(2, 0, 0)]
    for e in events:
        queue.push(e)
    before = tuple(queue._heap)
    assert queue.preview(3) == [events[2], events[3], events[1]]
    assert tuple(queue._heap) == before
    assert queue.peek() is events[2] and len(queue) == 4
    assert queue.preview(0) == []
    with pytest.raises(ValueError):
        queue.preview(-1)

    stepped = FortuneSweep(points, debug=True).start()
    previous = math.inf
    while (event := stepped.step()) is not None:
        assert event.y <= previous
        previous = event.y
        stepped.beach.validate()
    assert signature(stepped) == signature(batch)
    assert stepped.step() is None
    assert signature(stepped.run()) == signature(batch)

def test_browser_presentation_geometry_matches_python():

        pytest.skip("Node is only needed for the JavaScript presentation cross-check")

    cases = []
    for _ in range(100):
        a, b = Point(rng.random(), rng.random()), Point(rng.random(), rng.random())

const cases = JSON.parse(require('fs').readFileSync(0, 'utf8'));
console.log(JSON.stringify(cases.map(([a,b,d]) => [
  f.breakpointX(a,b,d), f.parabolaY(a,.31,d)
])));
"""
    result = subprocess.run(
        [node, "-e", command, script],
        input=json.dumps(cases),
        text=True,

        check=True,
    )
    for (a, b, d), (x, y) in zip(cases, json.loads(result.stdout)):

"""Independent O(n) / O(n^2) validators; never imported by the algorithm."""

import math
import random

from fortune.geometry import Point, polygon_area


def contains(polygon, p, tolerance=2e-10):
    if len(polygon) < 3:
        return False
    if (
        p.x < min(v.x for v in polygon) - tolerance
        or p.x > max(v.x for v in polygon) + tolerance
        or p.y < min(v.y for v in polygon) - tolerance
        or p.y > max(v.y for v in polygon) + tolerance
    ):
        return False
    coordinate_ulp = math.ulp(max(max(abs(v.x), abs(v.y)) for v in polygon))
    # Clipped cells are convex and ordered counterclockwise.
    for a, b in zip(polygon, polygon[1:] + polygon[:1]):
        cross = (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x)
        # Very short exported edges suffer endpoint subtraction error; the
        # cross-product uncertainty depends on coordinate ULP, not just length.
        ulp = coordinate_ulp
        uncertainty = 8 * ulp * (abs(p.x - a.x) + abs(p.y - a.y))
        if cross < -tolerance * max(1e-30, a.distance(b)) - uncertainty:
            return False
    return True


def validate_cells(diagram, queries=300, seed=42):
    box, rng = diagram.box, random.Random(seed)
    scale = max(box.xmax - box.xmin, box.ymax - box.ymin)
    area = sum(polygon_area(c.vertices) for c in diagram.cells)
    assert math.isclose(
        area,
        (box.xmax - box.xmin) * (box.ymax - box.ymin),
        rel_tol=2e-8,
        abs_tol=1e-11 * scale**2,
    ), ("area", area)
    for _ in range(queries):
        p = Point(rng.uniform(box.xmin, box.xmax), rng.uniform(box.ymin, box.ymax))
        owners = [
            c.site for c in diagram.cells if contains(c.vertices, p, 1e-10 * scale)
        ]
        distances = [p.distance(s) for s in diagram.sites]
        nearest = min(distances)
        assert owners, ("hole", p)
        assert all(distances[i] <= nearest + 1e-8 * scale for i in owners), (
            "owner",
            p,
            owners,
        )


def validate_empty_circles(diagram):
    for vertex in diagram.vertices:
        ds = [vertex.point.distance(p) for p in diagram.sites]
        defining = [ds[i] for i in vertex.defining]
        r = defining[0]
        # Exporting normalized results back near 1e9 necessarily rounds to a
        # world-coordinate ULP (~1e-7). Include that representational error.
        coordinate_ulp = math.ulp(max(abs(vertex.point.x), abs(vertex.point.y)))
        tolerance = 1e-8 * max(1, r) + 8 * coordinate_ulp
        assert max(defining) - min(defining) <= tolerance, ("equidistance", vertex)
        assert min(ds) >= r - tolerance, ("nonempty", vertex, min(ds), r)

    labels = {
        name: (number, int(page))
        for name, number, page in re.findall(
            r"\\newlabel\{([^}]+)\}\{\{([^}]+)\}\{(\d+)\}", aux
        )
    }
    body = labels["last-main-page"][1]
    text = [page.extract_text() for page in reader.pages]
    fonts = {}
    for page in reader.pages:
        resources = page["/Resources"]
        font_refs = resources.get("/Font")
        if font_refs is None:
            continue
        for ref in font_refs.get_object().values():
            font = ref.get_object()
            children = font.get("/DescendantFonts", [font])
            for child in children:
                obj = child.get_object()
                descriptor = obj.get("/FontDescriptor", {})
                if hasattr(descriptor, "get_object"):
                    descriptor = descriptor.get_object()
                embedded = any(
                    k in descriptor for k in ("/FontFile", "/FontFile2", "/FontFile3")
                )
                fonts[str(obj.get("/BaseFont", font.get("/BaseFont")))] = embedded
    all_a4 = all(
        all(
            abs(a - b) < 1
            for a, b in zip(
                sorted(map(float, (p.mediabox.width, p.mediabox.height))),
                [595.276, 841.890],
            )
        )
        for p in reader.pages
    )
    unresolved = bool(re.search(r"\?\?|\b(?:TODO|TBD|PLACEHOLDER)\b", "\n".join(text)))
    build = json.loads((ROOT / "evidence/report-build.json").read_text())
    from PIL import Image

    embedded_images = set()
    for page in reader.pages:
        for item in page.images:
            stream = item.indirect_reference.get_object()
            # The high-level extractor can re-encode JPEGs. Decode the original
            # DCT stream to compare original pixels without another lossy save.
            if str(stream.get("/Filter")) == "/DCTDecode":
                picture = Image.open(io.BytesIO(stream.get_data())).convert("RGB")
            else:
                picture = item.image.convert("RGB")
            embedded_images.add(
                (picture.size, hashlib.sha256(picture.tobytes()).hexdigest())
            )
    screenshot_names = (
        "gui-native.png",
        "ai-interaction-1.png",
        "pi-environment.png",
        "pi-interaction-invariants.png",
        "ai-interaction-2.png",
        "pi-tiny-cell-code-continuation.png",
        "pi-interaction-tests.png",
        "pi-tests-summary.png",
        "canvas-preview.jpg",
    )
    screenshot_matches = {}
    for name in screenshot_names:
        with Image.open(ROOT / "evidence" / name) as original:
            picture = original.convert("RGB")
            key = (picture.size, hashlib.sha256(picture.tobytes()).hexdigest())
            screenshot_matches[name] = key in embedded_images
    summary = dict(
        actual_timestamp=datetime.now(timezone(timedelta(hours=8))).isoformat(),
        report_engine="LuaLaTeX",
        pdf_sha256=hashlib.sha256(pdf.read_bytes()).hexdigest(),
        pages=len(reader.pages),
        main_body_pages=body,
        section_pages={k: v[1] for k, v in labels.items() if k.startswith("sec:")},
        counts={
            prefix: sum(k.startswith(prefix + ":") for k in labels)
            for prefix in ("fig", "tab", "eq", "lst")

        a4_all_pages=all_a4,
        body_font="Times New Roman 12 pt",
        body_spacing="1.5 (setspace onehalfspacing)",
        code_font="Courier New 9.5 pt; line numbers 9 pt",
        fonts_embedded=fonts,
        searchable_text=all(len(t.strip()) > 10 for t in text),
        unresolved_placeholders=unresolved,
        overflow_warnings=build["overfull_box_warnings"],
        owner_review_complete=json.loads((ROOT / "owner-metadata.json").read_text())[
            "owner_review_complete"
        ],
        source="report/report.tex",
        editable_sections=len(list((ROOT / "report/sections").glob("*.tex"))),
        bibliography_style="IEEE numbered (ieeetr)",
        bibliography_entries=len(re.findall(r"\\bibcite\{", aux)),
        embedded_screenshots_match_original_pixels=screenshot_matches,
        visual_review="Pending inspection of the newly rendered pages",

    if args.render:
        import pypdfium2 as pdfium
        from PIL import Image, ImageDraw

        stamp = datetime.now(timezone(timedelta(hours=8)))
        out = ROOT / "tmp" / stamp.strftime("report-pages-%Y%m%d-%H%M%S")

"""Reproducible timing plus isolated Python-allocation/Windows RSS measurements."""
# ruff: noqa: E402
# Standalone execution sets the project source path before local imports.

import argparse
import csv
import ctypes
import gc
import json
import platform
import statistics
import subprocess

import time
import tracemalloc
from datetime import datetime, timedelta, timezone

def windows_memory():
    if sys.platform != "win32":
        return {}
    from ctypes import wintypes

    class Counters(ctypes.Structure):
        _fields_ = [("cb", wintypes.DWORD), ("PageFaultCount", wintypes.DWORD)] + [
            (name, ctypes.c_size_t)
            for name in (
                "PeakWorkingSetSize",
                "WorkingSetSize",
                "QuotaPeakPagedPoolUsage",
                "QuotaPagedPoolUsage",
                "QuotaPeakNonPagedPoolUsage",
                "QuotaNonPagedPoolUsage",
                "PagefileUsage",
                "PeakPagefileUsage",
            )
        ]

    kernel = ctypes.WinDLL("kernel32", use_last_error=True)
    kernel.GetCurrentProcess.restype = wintypes.HANDLE
    psapi = ctypes.WinDLL("psapi", use_last_error=True)
    psapi.GetProcessMemoryInfo.argtypes = [
        wintypes.HANDLE,
        ctypes.POINTER(Counters),
        wintypes.DWORD,
    ]
    counters = Counters()
    counters.cb = ctypes.sizeof(counters)
    if not psapi.GetProcessMemoryInfo(
        kernel.GetCurrentProcess(), ctypes.byref(counters), counters.cb
    ):
        raise ctypes.WinError(ctypes.get_last_error())
    return dict(
        process_peak_rss_bytes=counters.PeakWorkingSetSize,
        process_working_set_bytes=counters.WorkingSetSize,
    )


def memory_sample(n, seed):
    points = random_sites(n, seed)
    gc.collect()
    tracemalloc.start()
    d = construct(points, Box(0, 0, 1, 1))
    current, peak = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    return dict(
        n=n,
        python_peak_bytes=peak,
        python_retained_bytes=current,
        cells=len(d.cells),
        **windows_memory(),
    )


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--sizes", type=int, nargs="+", default=[100, 1000, 10000])
    p.add_argument("--repeat", type=int, default=3)
    p.add_argument("--seed", type=int, default=2026)
    p.add_argument("--memory-child", type=int)
    p.add_argument("--out", default=str(ROOT / "results"))
    args = p.parse_args()
    if args.memory_child:
        print(json.dumps(memory_sample(args.memory_child, args.seed)))
        return
    if args.repeat < 1 or min(args.sizes) < 1:
        p.error("Sizes and repetition count must be positive")
    out = Path(args.out)
    out.mkdir(parents=True, exist_ok=True)
    rows, raw = [], []
    construct(random_sites(40, args.seed), Box(0, 0, 1, 1))
    for n in args.sizes:
        points = random_sites(n, args.seed)
        elapsed = []
        for repetition in range(args.repeat):
            gc.collect()
            started = time.perf_counter()
            d = construct(points, Box(0, 0, 1, 1))
            seconds = time.perf_counter() - started
            elapsed.append(seconds)
            raw.append(
                dict(
                    n=n,
                    repetition=repetition,
                    seconds=seconds,
                    sweep_seconds=d.statistics["sweep_seconds"],
                    vertices=len(d.vertices),
                    edges=len(d.edges),
                )
            )
            del d
        memory = json.loads(
            subprocess.check_output(
                [
                    sys.executable,
                    __file__,
                    "--memory-child",
                    str(n),
                    "--seed",
                    str(args.seed),
                ],
                text=True,
            )
        )
        row = dict(
            median_seconds=statistics.median(elapsed),
            min_seconds=min(elapsed),
            max_seconds=max(elapsed),
            **memory,
        )
        rows.append(row)
        print(json.dumps(row), flush=True)
    meta = dict(
        actual_timestamp=datetime.now(timezone(timedelta(hours=8))).isoformat(),
        python=sys.version,
        platform=platform.platform(),
        machine=platform.machine(),
        processor=platform.processor(),
        seed=args.seed,
        repetitions=args.repeat,
        timing_scope="deduplication, normalization, Fortune sweep, clipping, cell/edge assembly; excludes input generation, plotting, validation and export",
        memory_method="fresh subprocess per size; tracemalloc starts after input generation; process RSS includes interpreter and tracing overhead",
        results=rows,
        raw_timings=raw,
    )
    (out / "benchmark.json").write_text(json.dumps(meta, indent=2), encoding="utf-8")
    with (out / "benchmark.csv").open("w", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=rows[0].keys())
        writer.writeheader()
        writer.writerows(rows)


if __name__ == "__main__":
    main()

"""Compile editable LaTeX with native TeX or an existing WSL installation.

LaTeX section files are authoritative; Markdown is never regenerated or read.
Shell escape is disabled and actual compiler logs are retained.

def checked(command, *, cwd=None, timeout=300):
    return subprocess.run(
        command,
        cwd=cwd,
        text=True,
        encoding="utf-8",
        errors="replace",
        stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT,
        timeout=timeout,
        check=False,
    )


def select_engine(requested, distro):
    names = [requested] if requested != "auto" else ["lualatex", "xelatex"]
    if distro is None:
        for name in names:
            executable = shutil.which(name)
            if (
                executable
                and checked([executable, "--version"], timeout=20).returncode == 0
            ):
                return executable, None
        if os.name != "nt":
            raise RuntimeError("Install LuaLaTeX or XeLaTeX, or specify --engine.")
        distro = "Debian"
    for name in names:
        probe = checked(["wsl.exe", "-d", distro, "--", "which", name], timeout=30)
        if probe.returncode == 0:
            return name, distro
    raise RuntimeError(f"No working LuaLaTeX/XeLaTeX found in WSL {distro}.")


def wsl_path(path, distro):
    result = checked(
        ["wsl.exe", "-d", distro, "--", "wslpath", "-u", str(path)], timeout=30
    )
    if result.returncode:
        raise RuntimeError(result.stdout)
    return result.stdout.strip()

            font_cache.mkdir(exist_ok=True)
            local_path = wsl_path(local_tex, distro)
            cache_path = wsl_path(font_cache, distro)
            command.extend(
                [
                    "env",
                    f"TEXMFHOME={local_path}",
                    f"TEXMFVAR={cache_path}",
                    f"TEXMFCACHE={cache_path}",
                ]
            )
        command.append(engine)
        bib_command = [
            "wsl.exe",
            "-d",
            distro,
            "--cd",
            build_path,
            "--",
            "bibtex",
            "report",
        ]
        build_option = build_path
    else:
        command = [engine]
        bibtex = shutil.which("bibtex")
        if not bibtex:
            raise RuntimeError("BibTeX is required for the numbered references.")
        bib_command = [bibtex, "report"]
        build_option = str(BUILD)
    command.extend(
        [
            "-interaction=nonstopmode",
            "-halt-on-error",
            "-file-line-error",
            "-no-shell-escape",
            "-synctex=1",
            f"-output-directory={build_option}",
            "report.tex",
        ]
    )
    shutil.copy2(REPORT / "references.bib", BUILD / "references.bib")
    logs = []

    def run_stage(name, stage_command, cwd):
        result = checked(stage_command, cwd=cwd)
        (BUILD / f"{name}.log").write_text(result.stdout, encoding="utf-8")
        logs.append(f"=== {name} ===\n{result.stdout}")
        (ROOT / "evidence/report-build.log").write_text(
            "\n".join(logs), encoding="utf-8"
        )
        if result.returncode:
            raise RuntimeError(
                f"{name} failed:\n" + "\n".join(result.stdout.splitlines()[-45:])
            )
        return result.stdout

    run_stage("latex-pass-1", command, REPORT)
    run_stage("bibtex", bib_command, BUILD)
    run_stage("latex-pass-2", command, REPORT)
    final_output = run_stage("latex-pass-3", command, REPORT)
    if re.search(
        r"(?:Citation .* undefined|There were undefined references)", final_output
    ):
        raise RuntimeError(
            "Unresolved citations or references; see evidence/report-build.log"
        )
    pdf = BUILD / "report.pdf"
    if not pdf.is_file():
        raise RuntimeError("LaTeX did not produce report.pdf")
    shutil.copy2(pdf, REPORT / "report.pdf")
    source_listing()
    summary = {
        "source": "report/report.tex",
        "editable_sections": "report/sections/",
        "engine": Path(engine).name,
        "wsl_distribution": distro,
        "bibliography_style": "ieeetr",
        "latex_passes": 3,
        "shell_escape": False,
        "overfull_box_warnings": len(re.findall(r"Overfull \\[hv]box", final_output)),
        "pdf_bytes": pdf.stat().st_size,
        "output": "report/report.pdf",
    }
    (ROOT / "evidence/report-build.json").write_text(
        json.dumps(summary, indent=2) + "\n", encoding="utf-8"
    )
    print(json.dumps(summary, indent=2))


if __name__ == "__main__":
    try:
        main()
    except (RuntimeError, OSError, subprocess.SubprocessError) as error:
        print(str(error), file=sys.stderr)
        raise SystemExit(1) from error

"""Generate a real replay frame and optional GIF for reviewing the feature branch."""
# ruff: noqa: E402

import argparse
import json
import sys
from datetime import datetime, timedelta, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))

from matplotlib.backends.backend_agg import FigureCanvasAgg
from matplotlib.figure import Figure
from PIL import Image

from fortune.geometry import Box
from fortune.io import read_sites
from fortune.replay import SweepReplay
from fortune.replay_view import PANEL_RECT, PLOT_RECT, draw_replay


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input", type=Path, default=ROOT / "examples/sites.txt")
    parser.add_argument(
        "--gif", action="store_true", help="Also write the offline GIF preview"
    )
    args = parser.parse_args()
    model = SweepReplay(read_sites(args.input), Box(0, 0, 1, 1))
    fig = Figure(figsize=(12, 7), dpi=80)
    canvas = FigureCanvasAgg(fig)
    axes = fig.add_axes(PLOT_RECT)
    panel = fig.add_axes(PANEL_RECT)
    status = fig.text(
        0.07,
        0.05,
        "Program-generated replay preview | same event engine as the app",
        fontsize=10,
    )
    frames = []
    selected = None
    png = ROOT / "examples/sweep-replay.png"
    gif = ROOT / "examples/sweep-replay.gif"

    def render_frame():
        draw_replay(model, axes, panel)
        status.set_text(
            f"Replay | {model.event_count} heap entries processed | sweep y = {model.sweep_y:.4g}"
        )
        if args.gif:
            canvas.draw()
            im = Image.frombuffer(
                "RGBA",
                canvas.get_width_height(),
                canvas.buffer_rgba(),
                "raw",
                "RGBA",
                0,
                1,
            )
            frames.append(
                im.convert("RGB").convert(
                    "P", palette=Image.Palette.ADAPTIVE, colors=192
                )
            )

    render_frame()
    while not model.done:
        event = model.tick(2)
        render_frame()
        if (
            selected is None
            and event
            and event.kind == 1
            and not model.last_stale
            and not model.done
        ):
            fig.savefig(png, dpi=160)
            selected = dict(
                event_count=model.event_count,
                site_events=model.sweep.statistics["site_events"],
                circle_events=model.sweep.statistics["circle_events"],
                sweep_y=model.sweep_y,
            )
    if selected is None:
        fig.savefig(png, dpi=160)
    if args.gif:
        durations = [180] * len(frames)
        durations[0] = 700
        durations[-1] = 1500
        frames[0].save(
            gif, save_all=True, append_images=frames[1:], duration=durations, loop=0
        )
    meta = dict(
        actual_timestamp=datetime.now(timezone(timedelta(hours=8))).isoformat(),
        input=str(args.input.relative_to(ROOT))
        if args.input.is_relative_to(ROOT)
        else args.input.name,
        generated_by="tools/make_replay_example.py using actual SweepReplay state",
        png="examples/sweep-replay.png",
        selected_frame=selected,
        gif="examples/sweep-replay.gif" if args.gif else None,
        frames=len(frames),
    )
    (ROOT / "evidence/animation-preview.json").write_text(
        json.dumps(meta, indent=2) + "\n", encoding="utf-8"
    )
    print(json.dumps(meta, indent=2))


if __name__ == "__main__":
    main()

"""Reproduce replay rendering CPU measurements; these are not desktop FPS."""
# ruff: noqa: E402

import argparse
import cProfile
import io
import json
import platform
import pstats
import statistics
import sys
import time
from datetime import datetime, timedelta, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "src"))

from matplotlib.backends.backend_agg import FigureCanvasAgg
from matplotlib.figure import Figure

from fortune.geometry import Box
from fortune.io import random_sites
from fortune.replay import SweepReplay
from fortune.replay_view import PANEL_RECT, PLOT_RECT, ReplayController


def make_controller(n, *, comparison_rects=False):
    fig = Figure(figsize=(12, 7))
    FigureCanvasAgg(fig)
    plot = (0.07, 0.22, 0.58, 0.69) if comparison_rects else PLOT_RECT
    panel = (0.71, 0.25, 0.27, 0.64) if comparison_rects else PANEL_RECT
    return ReplayController(
        fig,
        fig.add_axes(plot),
        fig.add_axes(panel),
        SweepReplay(random_sites(n, 2026), Box(0, 0, 1, 1)),
    )


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--output",
        type=Path,
        default=ROOT / "results/animation-preview/render-after.json",
    )
    args = parser.parse_args()
    profiler = cProfile.Profile()
    rows = []
    for n in (9, 50):
        ui = make_controller(n, comparison_rects=True)
        ui.redraw()
        geometry = []
        paint = []
        for _ in range(48):
            start = time.perf_counter()
            ui.model.tick(5)
            geometry.append(time.perf_counter() - start)
            start = time.perf_counter()
            if n == 9:
                profiler.enable()
            ui.redraw()
            if n == 9:
                profiler.disable()
            paint.append(time.perf_counter() - start)
        rows.append(
            dict(
                sites=n,
                frames=48,
                median_geometry_ms=1000 * statistics.median(geometry),
                median_redraw_ms=1000 * statistics.median(paint),
                p95_redraw_ms=1000 * sorted(paint)[45],
                median_frame_ms=1000
                * statistics.median([a + b for a, b in zip(geometry, paint)]),
            )
        )
        ui.close()
    scrub = []
    for n in (9, 50, 200):
        ui = make_controller(n)
        samples = []
        for position in (70, 20, 85, 35, 90, 45, 95, 60, 30, 80):
            start = time.perf_counter()
            ui.position.set_val(position)
            samples.append(time.perf_counter() - start)
        scrub.append(
            dict(
                sites=n,
                samples=len(samples),
                median_ms=1000 * statistics.median(samples),
                max_ms=1000 * max(samples),
            )
        )
        ui.close()
    stream = io.StringIO()
    pstats.Stats(profiler, stream=stream).strip_dirs().sort_stats(
        "cumulative"
    ).print_stats(20)
    result = dict(
        actual_timestamp=datetime.now(timezone(timedelta(hours=8))).isoformat(),
        python=sys.version,
        platform=platform.platform(),
        scope="Same 1200x700 Agg canvas, inputs, 48-frame sequence and axes geometry as render-before.json; CPU rendering, not end-to-end desktop FPS. Nine-site measurements include cProfile overhead in both versions; 50-site measurements are unprofiled.",
        results=rows,
        position_scrubbing=scrub,
    )
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
    args.output.with_suffix(".profile.txt").write_text(
        stream.getvalue(), encoding="utf-8"
    )
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    main()

"""Measure the actual Tk playback timer and canvas on this desktop."""
# ruff: noqa: E402

import json
import statistics
import sys
import time
from datetime import datetime, timedelta, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]

import matplotlib

matplotlib.use("TkAgg")
import matplotlib.pyplot as plt

from fortune.geometry import Box
from fortune.io import random_sites
from fortune.replay_view import launch


def main():
    ui = launch(
        random_sites(50, 2026),
        Box(0, 0, 1, 1),
        ROOT / "examples/sweep-replay-live.png",
        block=False,
    )
    starts = []
    draws = []
    ui.timer.remove_callback(ui.tick)

    def measured_tick():
        start = time.perf_counter()
        ui.tick()
        draws.append(time.perf_counter() - start)
        starts.append(start)

    ui.timer.add_callback(measured_tick)
    root = ui.figure.canvas.get_tk_widget().winfo_toplevel()
    result = {}

    def finish():
        ui.pause()
        intervals = [b - a for a, b in zip(starts, starts[1:])]
        if intervals:
            result.update(
                actual_timestamp=datetime.now(timezone(timedelta(hours=8))).isoformat(),
                sites=50,
                backend=matplotlib.get_backend(),
                callbacks=len(starts),
                observed_window_seconds=8.0,
                median_callback_interval_ms=1000 * statistics.median(intervals),
                median_callback_rate_hz=1 / statistics.median(intervals),
                median_tick_render_ms=1000 * statistics.median(draws),
                p95_tick_render_ms=1000 * sorted(draws)[int(0.95 * (len(draws) - 1))],
                scope="Actual visible Tk window, timer callbacks and blit submission; not a measurement of display scanout.",
                event_count=ui.model.event_count,
            )
            ui.save_frame()
        ui.close()
        plt.close(ui.figure)

    root.after(250, ui.toggle_play)
    root.after(8250, finish)
    plt.show()
    if not result:
        raise RuntimeError(
            "Native playback was interrupted before measurements completed"
        )
    (ROOT / "results/animation-preview/render-tk.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    print(json.dumps(result, indent=2))


if __name__ == "__main__":
    main()

"""Independent stress validation, including every 10,000-site circle vertex.

NumPy is used only as a validation utility to accelerate brute-force distance
scans. It is not imported by the Fortune construction engine.
"""
# ruff: noqa: E402
# Standalone execution sets the project source/test paths before local imports.

import argparse
import json
import random
import sys
import time
import traceback
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path[:0] = [str(ROOT / "src"), str(ROOT / "tests")]
import numpy as np
from validation import validate_cells, validate_empty_circles

from fortune import Box, construct


def check_all_vertices(diagram):
    sites = np.array([(p.x, p.y) for p in diagram.sites])
    for vertex in diagram.vertices:
        distances = np.hypot(sites[:, 0] - vertex.point.x, sites[:, 1] - vertex.point.y)
        defining = distances[list(vertex.defining)]
        r = float(defining[0])
        tol = 1e-8 * max(1, r)
        assert float(defining.max() - defining.min()) <= tol, "equidistance"
        assert float(distances.min()) >= r - tol, "nonempty circle"


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--seeds", type=int, default=300)
    p.add_argument("--largest", type=int, default=10000)
    args = p.parse_args()
    started = time.perf_counter()
    failures = []
    checked = 0
    for seed in range(args.seeds):
        rng = random.Random(seed)
        d = construct(
            [(rng.random(), rng.random()) for _ in range(10 + seed % 80)],
            Box(0, 0, 1, 1),
        )
        validate_cells(d, queries=25, seed=seed)
        validate_empty_circles(d)
        checked += 1
    # More than one cluster seed prevents a fortunate single example from
    # masquerading as a broad near-coincidence check.
    for exponent in (10, 12):
        for seed in range(20):
            rng = random.Random(seed)
            points = [(0, 0), (1, 0), (1, 1), (0, 1)] + [
                (0.4 + 10**-exponent * rng.random(), 0.6 + 10**-exponent * rng.random())
                for _ in range(6)
            ]
            try:
                d = construct(points, debug=True)
                validate_cells(d, queries=50, seed=seed)
                validate_empty_circles(d)
                checked += 1
            except Exception as exc:
                failures.append(
                    dict(
                        exponent=exponent,
                        seed=seed,
                        input=points,
                        error=repr(exc),
                        traceback=traceback.format_exc(),
                    )
                )
    large = []
    for n in (100, 1000, args.largest):
        rng = random.Random(2026)
        d = construct([(rng.random(), rng.random()) for _ in range(n)], Box(0, 0, 1, 1))
        validate_cells(d, queries=100, seed=2026)
        check_all_vertices(d)
        large.append(dict(n=n, vertices_checked=len(d.vertices), cell_queries=100))
    result = dict(
        status="passed" if not failures else "failed",
        random_configurations=args.seeds,
        cluster_cases=40,
        checked_configurations=checked,
        queries_per_random=25,
        largest_cases=large,
        failures=failures,
        seconds=time.perf_counter() - started,
    )
    (ROOT / "evidence" / "stress-final.json").write_text(
        json.dumps(result, indent=2), encoding="utf-8"
    )
    print(json.dumps(result, indent=2))
    return bool(failures)


if __name__ == "__main__":
    raise SystemExit(main())

    run(
        "delivered independent test suite",

    )
    run(

        tests_passed=len(cases) - skipped,

        test_failures=failures,
        source_and_test_sha256=hashes,
        source_hash_scope="Delivered src/fortune, tests, package configuration and launchers; excludes generated egg-info and caches",
        log="evidence/package-verification.log",
        junit="evidence/package-tests.xml",
    )
    (ROOT / "evidence/package-verification.json").write_text(
        json.dumps(summary, indent=2) + "\n", encoding="utf-8"
    )
    print(
        json.dumps(
            {k: v for k, v in summary.items() if k != "source_and_test_sha256"},
            indent=2,
        )
    )
